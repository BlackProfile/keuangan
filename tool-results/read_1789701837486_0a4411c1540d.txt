    77→const DEFAULT_SETTINGS: AppSettings = {
    78→  pinEnabled: false,

    87→function parseSettings(raw: Record<string, string> | undefined): AppSettings {
    88→  if (!raw) return DEFAULT_SETTINGS;
    89→  return {
    90→    pinEnabled: raw.pinEnabled === "true",
    91→    pinHash: raw.pinHash ?? null,
    92→    hideAmounts: raw.hideAmounts === "true",
    93→    currency: raw.currency ?? "IDR",
    94→    reminderEnabled: raw.reminderEnabled === "true",
    95→    reminderHour: Number(raw.reminderHour ?? 20),
    96→    theme: (raw.theme as AppSettings["theme"]) ?? "system",
    97→  };
    98→}

   203→function KeamananSection({ settings }: { settings: AppSettings }) {
   204→  const updateMut = useUpdateSetting();
   205→  const [pinDialogOpen, setPinDialogOpen] = React.useState(false);
   206→  const [pin, setPin] = React.useState("");
   207→
   208→  function handleSwitchChange(enabled: boolean) {
   209→    if (enabled) {
   210→      setPin("");
   211→      setPinDialogOpen(true);
   212→    } else {
   213→      // Disable: clear pinHash and pinEnabled
   214→      updateMut.mutate({ key: "pinHash", value: "" });
   215→      updateMut.mutate({ key: "pinEnabled", value: "false" });
   216→      toast.success("PIN dinonaktifkan.");
   217→    }
   218→  }
   219→
   220→  async function handleSavePin() {
   221→    if (pin.length !== 4) {
   222→      toast.error("PIN harus 4 digit.");
   223→      return;
   224→    }
   225→    try {
   226→      const hash = await hashPin(pin);
   227→      updateMut.mutate({ key: "pinHash", value: hash });
   228→      updateMut.mutate({ key: "pinEnabled", value: "true" });
   229→      setPin("");
   230→      setPinDialogOpen(false);
   231→      toast.success("PIN aktif. Aplikasi akan terkunci saat dibuka.");
   232→    } catch {
   233→      toast.error("Gagal menghash PIN. Coba lagi.");
   234→    }
   235→  }
   236→
   237→  function toggleHideAmounts(enabled: boolean) {
   238→    updateMut.mutate({
   239→      key: "hideAmounts",
   240→      value: enabled ? "true" : "false",
   241→    });
   242→    toast.success(
   243→      enabled ? "Nominal disembunyikan." : "Nominal ditampilkan.",
   244→    );
   245→  }
   246→
   247→  return (
   248→    <SectionCard
   249→      icon={<ShieldCheck className="h-4 w-4" />}
   250→      title="Keamanan"
   251→      description="Lindungi privasi data keuangan Anda."
   252→    >
   253→      <SettingRow
   254→        icon={<Lock className="h-4 w-4" />}
   255→        title="Kunci PIN"
   256→        description="Minta PIN 4 digit saat membuka aplikasi."
   257→      >
   258→        <Switch
   259→          checked={settings.pinEnabled}
   260→          onCheckedChange={handleSwitchChange}
   261→          disabled={updateMut.isPending}
   262→          aria-label="Aktifkan kunci PIN"
   263→        />
   264→      </SettingRow>
   265→
   266→      <SettingRow
   267→        icon={<EyeOff className="h-4 w-4" />}
   268→        title="Sembunyikan Nominal"
   269→        description="Sembunyikan jumlah uang di semua layar."
   270→      >
   271→        <Switch
   272→          checked={settings.hideAmounts}
   273→          onCheckedChange={toggleHideAmounts}
   274→          disabled={updateMut.isPending}
   275→          aria-label="Sembunyikan nominal"
   276→        />
   277→      </SettingRow>
   278→
   279→      <Dialog open={pinDialogOpen} onOpenChange={setPinDialogOpen}>
   280→        <DialogContent showCloseButton={false} className="max-w-sm">
   281→          <DialogHeader>
   282→            <DialogTitle>Buat PIN 4 Digit</DialogTitle>
   283→            <DialogDescription>
   284→              PIN ini akan diminta setiap kali aplikasi dibuka.
   285→            </DialogDescription>
   286→          </DialogHeader>
   287→          <div className="flex flex-col items-center gap-4 py-4">
   288→            <InputOTP
   289→              maxLength={4}
   290→              value={pin}
   291→              onChange={(v) => setPin(v)}
   292→              autoFocus
   293→            >
   294→              <InputOTPGroup>
   295→                <InputOTPSlot index={0} />
   296→                <InputOTPSlot index={1} />
   297→                <InputOTPSlot index={2} />
   298→                <InputOTPSlot index={3} />
   299→              </InputOTPGroup>
   300→            </InputOTP>
   301→            <p className="text-center text-xs text-muted-foreground">
   302→              Ingat PIN ini — tidak bisa dipulihkan jika lupa.
   303→            </p>
   304→          </div>
   305→          <DialogFooter>
   306→            <DialogClose asChild>
   307→              <Button variant="outline">Batal</Button>
   308→            </DialogClose>
   309→            <Button
   310→              onClick={handleSavePin}
   311→              disabled={pin.length !== 4 || updateMut.isPending}
   312→              className="gap-1.5"
   313→            >
   314→              {updateMut.isPending && (
   315→                <Loader2 className="h-4 w-4 animate-spin" />
   316→              )}
   317→              Simpan PIN
   318→            </Button>
   319→          </DialogFooter>
   320→        </DialogContent>
   321→      </Dialog>
   322→    </SectionCard>
   323→  );
   324→}

   369→                    ? "bg-background text-foreground shadow-sm"
   370→                    : "text-muted-foreground hover:text-foreground",
   371→                )}
   372→                aria-pressed={active}
   373→                aria-label={`Tema ${t.label}`}

   390→function PengingatSection({ settings }: { settings: AppSettings }) {
   391→  const updateMut = useUpdateSetting();
   392→  const [hour, setHour] = React.useState(String(settings.reminderHour));
   393→
   394→  React.useEffect(() => {
   395→    setHour(String(settings.reminderHour));
   396→  }, [settings.reminderHour]);
   397→
   398→  // Schedule in-app browser notifications while the app is open
   399→  React.useEffect(() => {
   400→    if (!settings.reminderEnabled) return;
   401→    if (typeof window === "undefined" || !("Notification" in window)) return;
   402→
   403→    const targetHour = settings.reminderHour;
   404→    const interval = setInterval(() => {
   405→      const now = new Date();
   406→      if (now.getHours() !== targetHour) return;
   407→      const nowMin = now.getMinutes();
   408→      // Fire once near the top of the hour (first 5 minutes)
   409→      if (nowMin > 5) return;
   410→      const key = `dompetku-reminder-${now.toDateString()}`;
   411→      try {
   412→        if (localStorage.getItem(key)) return;
   413→        localStorage.setItem(key, "1");
   414→        if (Notification.permission === "granted") {
   415→          new Notification("DompetKu", {
   416→            body: "Saatnya mencatat transaksi hari ini! 📒",
   417→          });
   418→        }
   419→      } catch {
   420→        // localStorage may be unavailable (private mode) — ignore
   421→      }
   422→    }, 60_000);
   423→
   424→    return () => clearInterval(interval);
   425→  }, [settings.reminderEnabled, settings.reminderHour]);
   426→
   427→  async function handleToggleReminder(enabled: boolean) {
   428→    if (enabled && typeof window !== "undefined" && "Notification" in window) {
   429→      const perm = await Notification.requestPermission();
   430→      if (perm !== "granted") {
   431→        toast.warning(
   432→          "Izin notifikasi ditolak. Pengingat aktif tapi tanpa notifikasi.",
   433→        );
   434→      }
   435→    }
   436→    updateMut.mutate({
   437→      key: "reminderEnabled",
   438→      value: enabled ? "true" : "false",
   439→    });
   440→    toast.success(enabled ? "Pengingat aktif." : "Pengingat dinonaktifkan.");
   441→  }
   442→
   443→  function handleHourChange(v: string) {
   444→    setHour(v);
   445→    const num = Number(v);
   446→    if (Number.isInteger(num) && num >= 0 && num <= 23) {
   447→      updateMut.mutate({ key: "reminderHour", value: String(num) });
   448→    }
   449→  }
   450→
   451→  function handleTestNotification() {
   452→    if (typeof window === "undefined" || !("Notification" in window)) {
   453→      toast.error("Browser tidak mendukung notifikasi.");
   454→      return;
   455→    }
   456→    if (Notification.permission !== "granted") {
   457→      toast.warning("Berikan izin notifikasi dulu.");
   458→      return;
   459→    }
   460→    new Notification("DompetKu", {
   461→      body: "Ini contoh pengingat. 📒",
   462→    });
   463→    toast.success("Notifikasi tes dikirim.");
   464→  }
   465→
   466→  return (
   467→    <SectionCard
   468→      icon={<Bell className="h-4 w-4" />}
   469→      title="Pengingat"
   470→      description="Pengingat harian untuk mencatat transaksi."
   471→    >
   472→      <SettingRow
   473→        icon={<Clock className="h-4 w-4" />}
   474→        title="Pengingat Catat Harian"
   475→        description="Aktifkan pengingat untuk mencatat setiap hari."
   476→      >
   477→        <Switch
   478→          checked={settings.reminderEnabled}
   479→          onCheckedChange={handleToggleReminder}
   480→          disabled={updateMut.isPending}
   481→          aria-label="Aktifkan pengingat"
   482→        />
   483→      </SettingRow>
   484→
   485→      {settings.reminderEnabled && (
   486→        <>
   487→          <SettingRow
   488→            title="Waktu Pengingat"
   489→            description="Pengingat akan muncul setiap hari pada jam ini."
   490→          >
   491→            <div className="flex items-center gap-2">
   492→              <Input
   493→                type="number"
   494→                min={0}
   495→                max={23}
   496→                value={hour}
   497→                onChange={(e) => handleHourChange(e.target.value)}
   498→                className="h-9 w-16 text-center"
   499→                aria-label="Jam pengingat"
   500→              />
   501→              <span className="text-xs text-muted-foreground">:00</span>
   502→            </div>
   503→          </SettingRow>
   504→
   505→          <SettingRow
   506→            title="Uji Notifikasi"
   507→            description="Kirim notifikasi tes untuk memastikan izin aktif."
   508→          >
   509→            <Button
   510→              variant="outline"
   511→              size="sm"
   512→              onClick={handleTestNotification}
   513→              className="gap-1.5"
   514→            >
   515→              <Send className="h-3.5 w-3.5" />
   516→              Tes
   517→            </Button>
   518→          </SettingRow>
   519→        </>
   520→      )}
   521→    </SectionCard>
   522→  );
   523→}

   529→function DataSection() {
   530→  const importMut = useImportCsv();
   531→  const seedMut = useSeed();
   532→  const [importOpen, setImportOpen] = React.useState(false);
   533→  const [csvRows, setCsvRows] = React.useState<Array<Record<string, string>>>(
   534→    [],
   535→  );
   536→  const [csvHeaders, setCsvHeaders] = React.useState<string[]>([]);
   537→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   538→
   539→  function handleExportCsv() {
   540→    window.location.href = api.exportTransactionsUrl();
   541→    toast.success("Mengekspor CSV...");
   542→  }
   543→
   544→  function handleBackup() {
   545→    window.location.href = api.backupUrl();
   546→    toast.success("Membuat backup JSON...");
   547→  }
   548→
   549→  function handleImportClick() {
   550→    setCsvRows([]);
   551→    setCsvHeaders([]);
   552→    setImportOpen(true);
   553→  }
   554→
   555→  function handleFile(e: React.ChangeEvent<HTMLInputElement>) {
   556→    const file = e.target.files?.[0];
   557→    if (!file) return;
   558→    const reader = new FileReader();
   559→    reader.onload = () => {
   560→      const text = String(reader.result || "");
   561→      const { headers, rows } = parseCsv(text);
   562→      if (rows.length === 0) {
   563→        toast.error("CSV kosong atau tidak valid.");
   564→        return;
   565→      }
   566→      setCsvHeaders(headers);
   567→      setCsvRows(rows);
   568→    };
   569→    reader.onerror = () => toast.error("Gagal membaca file CSV.");
   570→    reader.readAsText(file);
   571→  }
   572→
   573→  function handleImport() {
   574→    if (csvRows.length === 0) return;
   575→    importMut.mutate(csvRows, {
   576→      onSuccess: (res) => {
   577→        toast.success(
   578→          `${res.imported} transaksi diimpor, ${res.skipped} dilewati.`,
   579→        );
   580→        setImportOpen(false);
   581→        setCsvRows([]);
   582→        setCsvHeaders([]);
   583→        if (fileInputRef.current) fileInputRef.current.value = "";
   584→      },
   585→      onError: (err) => {
   586→        toast.error(err.message || "Gagal mengimpor CSV.");
   587→      },
   588→    });
   589→  }
   590→
   591→  function handleCloseImport() {
   592→    setImportOpen(false);
   593→    setCsvRows([]);
   594→    setCsvHeaders([]);
   595→    if (fileInputRef.current) fileInputRef.current.value = "";
   596→  }
   597→
   598→  return (
   599→    <SectionCard
   600→      icon={<Database className="h-4 w-4" />}
   601→      title="Data"
   602→      description="Kelola data transaksi Anda."
   603→    >
   604→      <SettingRow
   605→        icon={<Upload className="h-4 w-4" />}
   606→        title="Impor CSV"
   607→        description="Impor transaksi dari file CSV."
   608→      >
   609→        <Button
   610→          variant="outline"
   611→          size="sm"
   612→          onClick={handleImportClick}
   613→          className="gap-1.5"
   614→        >
   615→          <Upload className="h-3.5 w-3.5" />
   616→          Impor
   617→        </Button>
   618→      </SettingRow>
   619→
   620→      <SettingRow
   621→        icon={<Download className="h-4 w-4" />}
   622→        title="Ekspor CSV"
   623→        description="Unduh semua transaksi sebagai CSV."
   624→      >
   625→        <Button
   626→          variant="outline"
   627→          size="sm"
   628→          onClick={handleExportCsv}
   629→          className="gap-1.5"
   630→        >
   631→          <Download className="h-3.5 w-3.5" />
   632→          Ekspor
   633→        </Button>
   634→      </SettingRow>
   635→
   636→      <SettingRow
   637→        icon={<FileJson className="h-4 w-4" />}
   638→        title="Backup JSON"
   639→        description="Cadangkan seluruh data (kategori, transaksi, dll)."
   640→      >
   641→        <Button
   642→          variant="outline"
   643→          size="sm"
   644→          onClick={handleBackup}
   645→          className="gap-1.5"

   795→// ============================================================
   796→// CSV parser (handles quoted fields with commas)
   797→// ============================================================
   798→
   799→function parseCsv(text: string): {
   800→  headers: string[];
   801→  rows: Array<Record<string, string>>;
   802→} {
   803→  // Strip BOM if present
   804→  const cleaned = text.replace(/^\uFEFF/, "");
   805→  const lines = cleaned
   806→    .split(/\r?\n/)
   807→    .map((l) => l.trim())
   808→    .filter((l) => l !== "");
   809→  if (lines.length === 0) return { headers: [], rows: [] };
   810→
   811→  const splitLine = (line: string): string[] => {
   812→    const out: string[] = [];
   813→    let cur = "";
   814→    let inQuotes = false;
   815→    for (let i = 0; i < line.length; i++) {
   816→      const ch = line[i];
   817→      if (ch === '"') {
   818→        if (inQuotes && line[i + 1] === '"') {
   819→          cur += '"';
   820→          i++;
   821→        } else {
   822→          inQuotes = !inQuotes;
   823→        }
   824→      } else if (ch === "," && !inQuotes) {
   825→        out.push(cur);
   826→        cur = "";
   827→      } else {
   828→        cur += ch;
   829→      }
   830→    }
   831→    out.push(cur);
   832→    return out.map((s) => s.trim());
   833→  };
   834→
   835→  const headers = splitLine(lines[0]);
   836→  const rows: Array<Record<string, string>> = [];
   837→  for (let i = 1; i < lines.length; i++) {
   838→    const cells = splitLine(lines[i]);
   839→    const obj: Record<string, string> = {};
   840→    headers.forEach((h, idx) => {
   841→      obj[h] = cells[idx] ?? "";
   842→    });
   843→    rows.push(obj);
   844→  }
   845→  return { headers, rows };
   846→}
