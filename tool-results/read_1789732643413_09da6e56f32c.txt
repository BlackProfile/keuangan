   523→              return (
   524→                <motion.div
   525→                  key={day}
   526→                  initial={{ opacity: 0, y: 8 }}
   527→                  animate={{ opacity: 1, y: 0 }}
   528→                  exit={{ opacity: 0, y: -8 }}
   529→                  transition={{ duration: 0.2 }}
   530→                  className="space-y-2"
   531→                >
   532→                  <div className="flex items-center justify-between px-1">
   533→                    <span className="flex items-center gap-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   534→                      {day}
   535→                      {hasPinned && (
   536→                        <Pin className="h-3 w-3 fill-primary text-primary" />
   537→                      )}
   538→                    </span>
   539→                    <div className="flex items-center gap-2 text-xs">
   540→                      {dayIncome > 0 && (
   541→                        <span className="font-medium text-income">
   542→                          +{formatCurrency(dayIncome)}
   543→                        </span>
   544→                      )}
   545→                      {dayExpense > 0 && (
   546→                        <span className="font-medium text-expense">
   547→                          −{formatCurrency(dayExpense)}
   548→                        </span>
   549→                      )}
   550→                      <Badge
   551→                        variant="secondary"
   552→                        className="text-[10px] font-normal"
   553→                      >
   554→                        {items.length}
   555→                      </Badge>
   556→                    </div>
   557→                  </div>
   558→                  <Card className="divide-y divide-border overflow-hidden p-0">
   559→                    {items.map((t) => (
   560→                      <TransactionRow
   561→                        key={t.id}
   562→                        transaction={t}
   563→                        onEdit={() => onEdit(t)}
   564→                      />
   565→                    ))}
   566→                  </Card>
   567→                </motion.div>
   568→              );
   569→            })}
   570→          </AnimatePresence>
   571→        </div>
   572→      )}
   573→    </div>
   574→  );
   575→}

   581→function SummaryStat({
   582→  label,
   583→  value,
   584→  variant,
   585→  loading,
   586→}: {
   587→  label: string;
   588→  value: number;
   589→  variant: "income" | "expense";
   590→  loading?: boolean;
   591→}) {
   592→  const isIncome = variant === "income";
   593→  return (
   594→    <Card className="p-3 sm:p-4">
   595→      <p className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground sm:text-xs">
   596→        {label}
   597→      </p>
   598→      {loading ? (
   599→        <Skeleton className="mt-1.5 h-5 w-20" />
   600→      ) : (
   601→        <p
   602→          className={cn(
   603→            "mt-1 text-sm font-bold tabular-nums sm:text-base",
   604→            isIncome ? "text-income" : "text-expense"
   605→          )}
   606→        >
   607→          {value < 0 && !isIncome ? "−" : ""}
   608→          {value >= 0 && isIncome ? "+" : ""}
   609→          {formatCurrency(Math.abs(value))}
   610→        </p>
   611→      )}
   612→    </Card>
   613→  );
   614→}

   616→function SummaryCount({
   617→  label,
   618→  value,
   619→  loading,
   620→}: {
   621→  label: string;
   622→  value: number;
   623→  loading?: boolean;
   624→}) {
   625→  return (
   626→    <Card className="p-3 sm:p-4">
   627→      <p className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground sm:text-xs">
   628→        {label}
   629→      </p>
   630→      {loading ? (
   631→        <Skeleton className="mt-1.5 h-5 w-12" />
   632→      ) : (
   633→        <p className="mt-1 text-sm font-bold tabular-nums text-foreground sm:text-base">

   667→function TransactionRow({
   668→  transaction,
   669→  onEdit,
   670→}: {
   671→  transaction: Transaction;
   672→  onEdit: () => void;
   673→}) {
   674→  const duplicateMut = useDuplicateTransaction();
   675→  const pinMut = useTogglePin();
   676→
   677→  const isIncome = transaction.type === "INCOME";
   678→  const cat = transaction.category;
   679→
   680→  const moodOpt = transaction.mood
   681→    ? MOOD_OPTIONS.find((o) => o.value === transaction.mood)
   682→    : null;
   683→  const priorityOpt = transaction.priority
   684→    ? PRIORITY_OPTIONS.find((o) => o.value === transaction.priority)
   685→    : null;
   686→  const paymentOpt = transaction.paymentMethod
   687→    ? PAYMENT_METHOD_OPTIONS.find((o) => o.value === transaction.paymentMethod)
   688→    : null;
   689→
   690→  const tags = parseTags(transaction.tags);
   691→
   692→  const flags: Array<{ label: string; color: string }> = [];
   693→  if (transaction.isDebt) flags.push({ label: "Hutang", color: "#ef4444" });
   694→  if (transaction.isReimbursable)
   695→    flags.push({ label: "Reimbursable", color: "#0891b2" });
   696→  if (transaction.isSubscription)
   697→    flags.push({ label: "Langganan", color: "#a855f7" });
   698→  if (transaction.isBusinessExpense)
   699→    flags.push({ label: "Bisnis", color: "#f97316" });
   700→  if (transaction.isTaxDeductible)
   701→    flags.push({ label: "Pajak", color: "#6b7280" });
   702→  if (transaction.isSplit) flags.push({ label: "Split", color: "#14b8a6" });
   703→
   704→  const hasFlagsRow = flags.length > 0 || tags.length > 0 || !!transaction.photoUrl;
   705→
   706→  function handleDuplicate() {
   707→    duplicateMut.mutate(transaction.id, {
   708→      onSuccess: () => toast.success("Transaksi berhasil diduplikat."),
   709→      onError: (e) => toast.error(e.message ?? "Gagal menduplikat transaksi."),
   710→    });
   711→  }
   712→
   713→  function handleTogglePin() {
   714→    pinMut.mutate(transaction.id, {
   715→      onSuccess: (updated) =>
   716→        toast.success(
   717→          updated.isPinned
   718→            ? "Transaksi disematkan."
   719→            : "Sematan dilepas."
   720→        ),
   721→      onError: (e) =>
   722→        toast.error(e.message ?? "Gagal mengubah sematan transaksi."),
   723→    });
   724→  }
   725→
   726→  return (
   727→    <div className="group flex items-center gap-3 px-3 py-3 transition-colors hover:bg-muted/40 sm:px-4">
   728→      {/* Category icon (with optional pinned ring) */}
   729→      <span
   730→        className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl"
   731→        style={{ backgroundColor: cat ? `${cat.color}1a` : undefined }}
   732→      >
   733→        <LucideIcon
   734→          name={cat?.icon ?? "Circle"}
   735→          className="h-5 w-5"
   736→          style={{ color: cat?.color }}
   737→        />
   738→      </span>
   739→
   740→      {/* Main content */}
   741→      <div className="min-w-0 flex-1">
   742→        {/* Title row: description + mood emoji + priority dot + payment icon + pin */}
   743→        <div className="flex items-center gap-1.5">
   744→          <p className="truncate text-sm font-medium text-foreground">
   745→            {transaction.description}
   746→          </p>
   747→          {moodOpt && (
   748→            <span
   749→              className="shrink-0 text-xs"
   750→              title={`Mood: ${moodOpt.label}`}
   751→              aria-label={`Mood: ${moodOpt.label}`}
   752→            >
   753→              {moodOpt.emoji}
   754→            </span>
   755→          )}
   756→          {priorityOpt && (
   757→            <span
   758→              className="h-1.5 w-1.5 shrink-0 rounded-full"
   759→              style={{ backgroundColor: priorityOpt.color }}
   760→              title={`Prioritas: ${priorityOpt.label}`}
   761→              aria-label={`Prioritas: ${priorityOpt.label}`}
   762→            />
   763→          )}
   764→          {paymentOpt && (
   765→            <LucideIcon
   766→              name={paymentOpt.icon}
   767→              className="h-3 w-3 shrink-0 text-muted-foreground"
   768→              aria-label={`Pembayaran: ${paymentOpt.label}`}
   769→            />
   770→          )}
   771→          {transaction.isPinned && (
   772→            <Pin
   773→              className="h-3 w-3 shrink-0 fill-primary text-primary"
   774→              aria-label="Disematkan"
   775→            />
   776→          )}
   777→        </div>
   778→
   779→        {/* Category + date row */}
   780→        <p className="mt-0.5 truncate text-xs text-muted-foreground">
   781→          {cat?.name ?? "Tanpa kategori"}
   782→          <span className="mx-1 text-border">·</span>
   783→          {formatDateLong(transaction.date)}
   784→          {transaction.account && (
   785→            <>
   786→              <span className="mx-1 text-border">·</span>
   787→              <LucideIcon
   788→                name={transaction.account.icon}
   789→                className="-mt-0.5 mr-0.5 inline h-3 w-3"
   790→                style={{ color: transaction.account.color }}
   791→              />
   792→              {transaction.account.name}
   793→            </>
   794→          )}
   795→        </p>
   796→
   797→        {/* Flags row */}
   798→        {hasFlagsRow && (
   799→          <div className="mt-1.5 flex flex-wrap items-center gap-1">
   800→            {flags.map((f) => (
   801→              <span
   802→                key={f.label}
   803→                className="inline-flex items-center rounded-full px-1.5 py-0.5 text-[10px] font-medium leading-none"
   804→                style={{
   805→                  color: f.color,
   806→                  backgroundColor: `${f.color}1a`,
   807→                }}
   808→              >
   809→                {f.label}
   810→              </span>
   811→            ))}
   812→            {tags.map((tag) => (
   813→              <span
   814→                key={tag}
   815→                className="inline-flex items-center rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-medium leading-none text-muted-foreground"
   816→              >
   817→                #{tag}
   818→              </span>
   819→            ))}
   820→            {transaction.photoUrl && (
   821→              <span
   822→                className="inline-flex items-center rounded-full bg-muted px-1.5 py-0.5 text-[10px] leading-none text-muted-foreground"
   823→                title="Memiliki foto"
   824→              >
   825→                <Camera className="h-3 w-3" />
   826→              </span>
   827→            )}
   828→          </div>
   829→        )}
   830→      </div>
   831→
   832→      {/* Right side: amount + actions */}
   833→      <div className="flex shrink-0 items-center gap-2">
   834→        <div className="text-right">
   835→          <div
   836→            className={cn(
   837→              "text-sm font-semibold tabular-nums",
   838→              isIncome ? "text-income" : "text-expense"
   839→            )}
   840→          >
   841→            {isIncome ? "+" : "−"}
   842→            {formatCurrency(transaction.amount)}
   843→          </div>
   844→        </div>
   845→        {/* Actions: combine edit + menu into single dropdown to avoid overlap */}
   846→        <DropdownMenu>
   847→          <DropdownMenuTrigger asChild>
   848→            <Button
   849→              variant="ghost"
   850→              size="icon"
   851→              className="h-8 w-8 shrink-0 text-muted-foreground opacity-60 transition-opacity hover:bg-muted hover:text-foreground focus-visible:opacity-100 group-hover:opacity-100"
   852→              aria-label="Aksi transaksi"
   853→            >
   854→              <MoreVertical className="h-4 w-4" />
   855→            </Button>
   856→          </DropdownMenuTrigger>
   857→          <DropdownMenuContent align="end" className="w-44">
   858→            <DropdownMenuItem
   859→              onClick={onEdit}
   860→              className="gap-2"
   861→            >
   862→              <Pencil className="h-3.5 w-3.5" />
   863→              Edit
   864→            </DropdownMenuItem>
   865→            <DropdownMenuItem
   866→              onClick={handleDuplicate}
   867→              disabled={duplicateMut.isPending}
   868→            >
   869→              {duplicateMut.isPending ? (
   870→                <Loader2 className="h-4 w-4 animate-spin" />
   871→              ) : (
   872→                <Copy className="h-4 w-4" />
   873→              )}
   874→              Duplikat
   875→            </DropdownMenuItem>
   876→            <DropdownMenuItem
   877→              onClick={handleTogglePin}
   878→              disabled={pinMut.isPending}
   879→            >
   880→              {pinMut.isPending ? (
   881→                <Loader2 className="h-4 w-4 animate-spin" />
   882→              ) : (
   883→                <Pin className="h-4 w-4" />
   884→              )}
   885→              {transaction.isPinned ? "Lepas Sematan" : "Sematkan"}
   886→            </DropdownMenuItem>
   887→            <DropdownMenuSeparator />
   888→            <DropdownMenuItem onClick={() => toast("Detail transaksi")}>
   889→              <Eye className="h-4 w-4" />
   890→              Lihat Detail
   891→            </DropdownMenuItem>
   892→          </DropdownMenuContent>
   893→        </DropdownMenu>
   894→      </div>
   895→    </div>
   896→  );
   897→}

   919→function useDebouncedValue<T>(value: T, delay: number): T {
   920→  const [debounced, setDebounced] = React.useState(value);
   921→  React.useEffect(() => {
   922→    const t = setTimeout(() => setDebounced(value), delay);
   923→    return () => clearTimeout(t);
   924→  }, [value, delay]);
   925→  return debounced;
   926→}
