   524→            </div>
   525→          )}
   526→        </Card>
   527→      )}
   528→
   529→      {/* List */}
   530→      {isLoading ? (
   531→        <div className="space-y-3">

   534→          ))}
   535→        </div>
   536→      ) : grouped.length === 0 ? (
   537→        <EmptyState title={emptyTitle} description={emptyDescription} />
   538→      ) : (
   539→        <div className="space-y-2">
   540→          <AnimatePresence mode="popLayout">
   541→            {grouped.map(([day, items]) => {
   542→              const dayIncome = items
   543→                .filter((t) => t.type === "INCOME")
   544→                .reduce((s, t) => s + t.amount, 0);
   545→              const dayExpense = items
   546→                .filter((t) => t.type === "EXPENSE")
   547→                .reduce((s, t) => s + t.amount, 0);
   548→              const hasPinned = items.some((t) => t.isPinned);
   549→              return (
   550→                <motion.div
   551→                  key={day}
   552→                  initial={{ opacity: 0, y: 4 }}
   553→                  animate={{ opacity: 1, y: 0 }}
   554→                  exit={{ opacity: 0, y: -4 }}
   555→                  transition={{ duration: 0.15 }}
   556→                  className="space-y-1"
   557→                >
   558→                  <div className="flex items-center justify-between px-1">
   559→                    <span className="flex items-center gap-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   560→                      {day}
   561→                      {hasPinned && (
   562→                        <Pin className="h-3 w-3 fill-primary text-primary" />
   563→                      )}
   564→                    </span>
   565→                    <div className="flex items-center gap-2 text-xs">
   566→                      {dayIncome > 0 && (
   567→                        <span className="font-medium text-income">
   568→                          +{formatCurrency(dayIncome)}
   569→                        </span>
   570→                      )}
   571→                      {dayExpense > 0 && (
   572→                        <span className="font-medium text-expense">
   573→                          −{formatCurrency(dayExpense)}
   574→                        </span>
   575→                      )}
   576→                      <Badge
   577→                        variant="secondary"
   578→                        className="text-[10px] font-normal"
   579→                      >
   580→                        {items.length}
   581→                      </Badge>
   582→                    </div>
   583→                  </div>
   584→                  <Card className="divide-y divide-border overflow-hidden p-0">
   585→                    {items.map((t) => (
   586→                      <TransactionRow
   587→                        key={t.id}
   588→                        transaction={t}
   589→                        onEdit={() => onEdit(t)}
   590→                      />
   591→                    ))}
   592→                  </Card>
   593→                </motion.div>
   594→              );
   595→            })}
   596→          </AnimatePresence>
   597→        </div>
   598→      )}
   599→    </div>
   600→  );
   601→}

   607→function SummaryStat({
   608→  label,
   609→  value,
   610→  variant,
   611→  loading,
   612→}: {
   613→  label: string;
   614→  value: number;
   615→  variant: "income" | "expense";
   616→  loading?: boolean;
   617→}) {
   618→  const isIncome = variant === "income";
   619→  return (
   620→    <Card className="p-3 sm:p-4">
   621→      <p className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground sm:text-xs">
   622→        {label}
   623→      </p>
   624→      {loading ? (
   625→        <Skeleton className="mt-1.5 h-5 w-20" />
   626→      ) : (
   627→        <p

   629→            "mt-1 text-sm font-bold tabular-nums sm:text-base",

   635→          {formatCurrency(Math.abs(value))}
   636→        </p>
   637→      )}
   638→    </Card>

   693→function TransactionRow({
   694→  transaction,
   695→  onEdit,
   696→}: {
   697→  transaction: Transaction;
   698→  onEdit: () => void;
   699→}) {
   700→  const duplicateMut = useDuplicateTransaction();
   701→  const pinMut = useTogglePin();
   702→  const hideMut = useToggleHideTransaction();
   703→
   704→  function handleToggleHide() {
   705→    hideMut.mutate(transaction.id, {
   706→      onSuccess: () => {
   707→        toast.success(transaction.isHidden ? "Transaksi ditampilkan" : "Transaksi disembunyikan");
   708→      },
   709→      onError: () => toast.error("Gagal mengubah status transaksi"),
   710→    });
   711→  }
   712→
   713→  const isIncome = transaction.type === "INCOME";
   714→  const cat = transaction.category;
   715→
   716→  const moodOpt = transaction.mood
   717→    ? MOOD_OPTIONS.find((o) => o.value === transaction.mood)
   718→    : null;
   719→  const priorityOpt = transaction.priority
   720→    ? PRIORITY_OPTIONS.find((o) => o.value === transaction.priority)
   721→    : null;
   722→  const paymentOpt = transaction.paymentMethod
   723→    ? PAYMENT_METHOD_OPTIONS.find((o) => o.value === transaction.paymentMethod)
   724→    : null;
   725→
   726→  const tags = parseTags(transaction.tags);
   727→
   728→  const flags: Array<{ label: string; color: string }> = [];
   729→  if (transaction.isDebt) flags.push({ label: "Hutang", color: "#ef4444" });
   730→  if (transaction.isReimbursable)
   731→    flags.push({ label: "Reimbursable", color: "#0891b2" });
   732→  if (transaction.isSubscription)
   733→    flags.push({ label: "Langganan", color: "#a855f7" });
   734→  if (transaction.isBusinessExpense)
   735→    flags.push({ label: "Bisnis", color: "#f97316" });
   736→  if (transaction.isTaxDeductible)
   737→    flags.push({ label: "Pajak", color: "#6b7280" });
   738→  if (transaction.isSplit) flags.push({ label: "Split", color: "#14b8a6" });
   739→
   740→  const hasFlagsRow = flags.length > 0 || tags.length > 0 || !!transaction.photoUrl;
   741→
   742→  function handleDuplicate() {
   743→    duplicateMut.mutate(transaction.id, {
   744→      onSuccess: () => toast.success("Transaksi berhasil diduplikat."),
   745→      onError: (e) => toast.error(e.message ?? "Gagal menduplikat transaksi."),
   746→    });
   747→  }
   748→
   749→  function handleTogglePin() {
   750→    pinMut.mutate(transaction.id, {
   751→      onSuccess: (updated) =>
   752→        toast.success(
   753→          updated.isPinned
   754→            ? "Transaksi disematkan."
   755→            : "Sematan dilepas."
   756→        ),
   757→      onError: (e) =>
   758→        toast.error(e.message ?? "Gagal mengubah sematan transaksi."),
   759→    });
   760→  }
   761→
   762→  return (
   763→    <div className="group flex items-center gap-3 px-3 py-2 transition-colors hover:bg-muted/40 sm:px-4">
   764→      {/* Category icon (with optional pinned ring) */}
   765→      <span
   766→        className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl"
   767→        style={{ backgroundColor: cat ? `${cat.color}1a` : undefined }}
   768→      >
   769→        <LucideIcon
   770→          name={cat?.icon ?? "Circle"}
   771→          className="h-4.5 w-4.5"
   772→          style={{ color: cat?.color }}
   773→        />
   774→      </span>
   775→
   776→      {/* Main content */}
   777→      <div className="min-w-0 flex-1">
   778→        {/* Title row: description + mood emoji + priority dot + payment icon + pin */}
   779→        <div className="flex items-center gap-1.5">
   780→          <p className="truncate text-sm font-medium text-foreground">
   781→            {transaction.description}
   782→          </p>
   783→          {moodOpt && (
   784→            <span
   785→              className="shrink-0 text-xs"
   786→              title={`Mood: ${moodOpt.label}`}
   787→              aria-label={`Mood: ${moodOpt.label}`}
   788→            >
   789→              {moodOpt.emoji}
   790→            </span>
   791→          )}
   792→          {priorityOpt && (
   793→            <span
   794→              className="h-1.5 w-1.5 shrink-0 rounded-full"
   795→              style={{ backgroundColor: priorityOpt.color }}
   796→              title={`Prioritas: ${priorityOpt.label}`}
   797→              aria-label={`Prioritas: ${priorityOpt.label}`}
   798→            />
   799→          )}
   800→          {paymentOpt && (
   801→            <LucideIcon
   802→              name={paymentOpt.icon}
   803→              className="h-3 w-3 shrink-0 text-muted-foreground"
   804→              aria-label={`Pembayaran: ${paymentOpt.label}`}
   805→            />
   806→          )}
   807→          {transaction.isPinned && (
   808→            <Pin
   809→              className="h-3 w-3 shrink-0 fill-primary text-primary"
   810→              aria-label="Disematkan"
   811→            />
   812→          )}
   813→        </div>
   814→
   815→        {/* Category + date row */}
   816→        <p className="mt-0.5 truncate text-xs text-muted-foreground">
   817→          {cat?.name ?? "Tanpa kategori"}
   818→          <span className="mx-1 text-border">·</span>
   819→          {relativeDay(transaction.date)}
   820→        </p>
   821→
   822→        {/* Flags row */}
   823→        {hasFlagsRow && (
   824→          <div className="mt-1.5 flex flex-wrap items-center gap-1">
   825→            {flags.map((f) => (
   826→              <span
   827→                key={f.label}
   828→                className="inline-flex items-center rounded-full px-1.5 py-0.5 text-[10px] font-medium leading-none"
   829→                style={{
   830→                  color: f.color,
   831→                  backgroundColor: `${f.color}1a`,
   832→                }}
   833→              >
   834→                {f.label}
   835→              </span>
   836→            ))}
   837→            {tags.map((tag) => (
   838→              <span
   839→                key={tag}
   840→                className="inline-flex items-center rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-medium leading-none text-muted-foreground"
   841→              >
   842→                #{tag}
   843→              </span>
   844→            ))}
   845→            {transaction.photoUrl && (
   846→              <span
   847→                className="inline-flex items-center rounded-full bg-muted px-1.5 py-0.5 text-[10px] leading-none text-muted-foreground"
   848→                title="Memiliki foto"
   849→              >
   850→                <Camera className="h-3 w-3" />
   851→              </span>
   852→            )}
   853→          </div>
   854→        )}
   855→      </div>
   856→
   857→      {/* Right side: amount + actions */}
   858→      <div className="flex shrink-0 items-center gap-2">
   859→        <div className="text-right">
   860→          <div
   861→            className={cn(
   862→              "text-sm font-semibold tabular-nums",
   863→              isIncome ? "text-income" : "text-expense"
   864→            )}
   865→          >
   866→            {isIncome ? "+" : "−"}
   867→            {formatCurrency(transaction.amount)}
   868→          </div>
   869→        </div>
   870→        {/* Actions: combine edit + menu into single dropdown to avoid overlap */}
   871→        <DropdownMenu>
   872→          <DropdownMenuTrigger asChild>
   873→            <Button
   874→              variant="ghost"
   875→              size="icon"
   876→              className="h-8 w-8 shrink-0 text-muted-foreground opacity-60 transition-opacity hover:bg-muted hover:text-foreground focus-visible:opacity-100 group-hover:opacity-100"
   877→              aria-label="Aksi transaksi"
   878→            >
   879→              <MoreVertical className="h-4 w-4" />
   880→            </Button>
   881→          </DropdownMenuTrigger>
   882→          <DropdownMenuContent align="end" className="w-44">
   883→            <DropdownMenuItem
   884→              onClick={onEdit}
   885→              className="gap-2"
   886→            >
   887→              <Pencil className="h-3.5 w-3.5" />
   888→              Edit
   889→            </DropdownMenuItem>
   890→            <DropdownMenuItem
   891→              onClick={handleDuplicate}
   892→              disabled={duplicateMut.isPending}
   893→            >
   894→              {duplicateMut.isPending ? (
   895→                <Loader2 className="h-4 w-4 animate-spin" />
   896→              ) : (
   897→                <Copy className="h-4 w-4" />
   898→              )}
   899→              Duplikat
   900→            </DropdownMenuItem>
   901→            <DropdownMenuItem
   902→              onClick={handleTogglePin}
   903→              disabled={pinMut.isPending}
   904→            >
   905→              {pinMut.isPending ? (
   906→                <Loader2 className="h-4 w-4 animate-spin" />
   907→              ) : (
   908→                <Pin className="h-4 w-4" />
   909→              )}
   910→              {transaction.isPinned ? "Lepas Sematan" : "Sematkan"}
   911→            </DropdownMenuItem>
   912→            <DropdownMenuItem
   913→              onClick={handleToggleHide}
   914→              disabled={hideMut.isPending}
   915→            >
   916→              {hideMut.isPending ? (
   917→                <Loader2 className="h-4 w-4 animate-spin" />
   918→              ) : transaction.isHidden ? (
   919→                <Eye className="h-4 w-4" />
   920→              ) : (
   921→                <EyeOff className="h-4 w-4" />
   922→              )}
   923→              {transaction.isHidden ? "Tampilkan" : "Sembunyikan"}
   924→            </DropdownMenuItem>
   925→            <DropdownMenuSeparator />
   926→            <DropdownMenuItem onClick={() => toast("Detail transaksi")}>
   927→              <Eye className="h-4 w-4" />
   928→              Lihat Detail
   929→            </DropdownMenuItem>
   930→          </DropdownMenuContent>
   931→        </DropdownMenu>
   932→      </div>
   933→    </div>
   934→  );
   935→}

   957→function useDebouncedValue<T>(value: T, delay: number): T {
   958→  const [debounced, setDebounced] = React.useState(value);
   959→  React.useEffect(() => {
   960→    const t = setTimeout(() => setDebounced(value), delay);
   961→    return () => clearTimeout(t);
   962→  }, [value, delay]);
   963→  return debounced;
   964→}
