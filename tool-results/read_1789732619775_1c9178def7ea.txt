   119→function bufferToBase64(buffer: ArrayBuffer): string {
   120→  const bytes = new Uint8Array(buffer);
   121→  let str = "";
   122→  for (let i = 0; i < bytes.length; i++) str += String.fromCharCode(bytes[i]);
   123→  return btoa(str);
   124→}

   126→function passwordStrength(
   127→  pw: string,
   128→): { score: 0 | 1 | 2 | 3 | 4; label: string; color: string } {
   129→  if (!pw) return { score: 0, label: "Kosong", color: "bg-muted" };
   130→  let score = 0;
   131→  if (pw.length >= 8) score++;
   132→  if (pw.length >= 12) score++;
   133→  if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) score++;
   134→  if (/[0-9]/.test(pw) && /[^a-zA-Z0-9]/.test(pw)) score++;
   135→  const labels = ["Sangat Lemah", "Lemah", "Sedang", "Kuat", "Sangat Kuat"];
   136→  const colors = [
   137→    "bg-rose-500",
   138→    "bg-rose-500",
   139→    "bg-amber-500",
   140→    "bg-emerald-500",
   141→    "bg-emerald-600",
   142→  ];
   143→  return {
   144→    score: score as 0 | 1 | 2 | 3 | 4,
   145→    label: labels[score],
   146→    color: colors[score],
   147→  };
   148→}

   150→function formatTimeAgo(iso: string): string {
   151→  const date = new Date(iso);
   152→  const diff = Date.now() - date.getTime();
   153→  const minutes = Math.floor(diff / 60_000);
   154→  if (minutes < 1) return "Baru saja";
   155→  if (minutes < 60) return `${minutes} menit lalu`;
   156→  const hours = Math.floor(minutes / 60);
   157→  if (hours < 24) return `${hours} jam lalu`;
   158→  const days = Math.floor(hours / 24);
   159→  if (days < 30) return `${days} hari lalu`;
   160→  return formatDateLong(iso);
   161→}

   291→export function SecuritySection() {
   292→  const { data: rawConfig, isLoading } = useSecuritySettings();
   293→  const bulkMut = useUpdateSecurityBulk();
   294→
   295→  const serverConfig = React.useMemo(
   296→    () => (rawConfig ? parseSecurityConfig(rawConfig) : DEFAULT_SECURITY_CONFIG),
   297→    [rawConfig],
   298→  );
   299→
   300→  const [localConfig, setLocalConfig] =
   301→    React.useState<SecurityConfig>(DEFAULT_SECURITY_CONFIG);
   302→  // Sync from server only on initial load to avoid overwriting unsaved local
   303→  // edits after a bulk save triggers a query refetch.
   304→  const hasInitializedRef = React.useRef(false);
   305→  const saveTimerRef = React.useRef<ReturnType<typeof setTimeout> | null>(null);
   306→  const mutateRef = React.useRef(bulkMut.mutate);
   307→  React.useEffect(() => {
   308→    mutateRef.current = bulkMut.mutate;
   309→  }, [bulkMut.mutate]);
   310→  const retryCountRef = React.useRef(0);
   311→
   312→  // Sync from server on first load (and when raw data first arrives)
   313→  React.useEffect(() => {
   314→    if (rawConfig && !hasInitializedRef.current) {
   315→      setLocalConfig(serverConfig);
   316→      hasInitializedRef.current = true;
   317→    }
   318→  }, [rawConfig, serverConfig]);
   319→
   320→  // Debounced bulk save (skipped until first server sync completes)
   321→  React.useEffect(() => {
   322→    if (!hasInitializedRef.current) return;
   323→    if (saveTimerRef.current) clearTimeout(saveTimerRef.current);
   324→    saveTimerRef.current = setTimeout(() => {
   325→      mutateRef.current(serializeSecurityConfig(localConfig), {
   326→        onSuccess: () => {
   327→          retryCountRef.current = 0;
   328→          toast.success("Pengaturan disimpan");
   329→        },
   330→        onError: (err) => {
   331→          // Retry up to 3 times with backoff for transient failures
   332→          if (retryCountRef.current < 3) {
   333→            retryCountRef.current += 1;
   334→            const delay = 1000 * retryCountRef.current;
   335→            setTimeout(() => {
   336→              mutateRef.current(serializeSecurityConfig(localConfig), {
   337→                onSuccess: () => {
   338→                  retryCountRef.current = 0;
   339→                  toast.success("Pengaturan disimpan");
   340→                },
   341→                onError: () => {
   342→                  if (retryCountRef.current >= 3) {
   343→                    toast.error("Gagal menyimpan — server tidak tersambung");
   344→                  }
   345→                },
   346→              });
   347→            }, delay);
   348→          } else {
   349→            toast.error("Gagal menyimpan — server tidak tersambung");
   350→          }
   351→        },
   352→      });
   353→    }, 500);
   354→    return () => {
   355→      if (saveTimerRef.current) clearTimeout(saveTimerRef.current);
   356→    };
   357→  }, [localConfig]);
   358→
   359→  const updateConfig = React.useCallback(
   360→    (partial: Partial<SecurityConfig>) => {
   361→      setLocalConfig((prev) => ({ ...prev, ...partial }));
   362→    },
   363→    [],
   364→  );
   365→
   366→  return (
   367→    <div className="space-y-4">
   368→      <div>
   369→        <h2 className="text-xl font-bold tracking-tight">Keamanan</h2>
   370→        <p className="mt-0.5 text-sm text-muted-foreground">
   371→          Konfigurasi perlindungan data keuangan Anda. Perubahan disimpan otomatis.
   372→        </p>
   373→      </div>
   374→
   375→      {isLoading ? (
   376→        <div className="space-y-4">
   377→          {Array.from({ length: 6 }).map((_, i) => (
   378→            <Skeleton key={i} className="h-40 w-full rounded-2xl" />
   379→          ))}
   380→        </div>
   381→      ) : (
   382→        <div className="space-y-4">
   383→          <AppLockSection config={localConfig} updateConfig={updateConfig} />
   384→          <RateLimitSection config={localConfig} updateConfig={updateConfig} />
   385→          <ReauthSection config={localConfig} updateConfig={updateConfig} />
   386→          <PrivacySection config={localConfig} updateConfig={updateConfig} />
   387→          <DecoySection config={localConfig} updateConfig={updateConfig} />
   388→          <SessionSection config={localConfig} updateConfig={updateConfig} />
   389→          <TrustedDevicesSection />
   390→          <NetworkSection config={localConfig} updateConfig={updateConfig} />
   391→          <EncryptionSection config={localConfig} updateConfig={updateConfig} />
   392→          <BackupSection config={localConfig} updateConfig={updateConfig} />
   393→          <AuditLogSection config={localConfig} updateConfig={updateConfig} />
   394→          <EmergencySection />
   395→        </div>
   396→      )}
   397→    </div>
   398→  );
   399→}

   593→function PinSetupDialog({
   594→  open,
   595→  onOpenChange,
   596→  onConfirm,
   597→}: {
   598→  open: boolean;
   599→  onOpenChange: (v: boolean) => void;
   600→  onConfirm: (pin: string) => void | Promise<void>;
   601→}) {
   602→  const [pin, setPin] = React.useState("");
   603→  const [confirm, setConfirm] = React.useState("");
   604→  const [submitting, setSubmitting] = React.useState(false);
   605→
   606→  React.useEffect(() => {
   607→    if (!open) {
   608→      setPin("");
   609→      setConfirm("");
   610→      setSubmitting(false);
   611→    }
   612→  }, [open]);
   613→
   614→  const validLength = pin.length >= 4 && pin.length <= 6;
   615→  const allDigits = /^\d*$/.test(pin) && pin.length > 0;
   616→  const match = pin.length > 0 && pin === confirm;
   617→
   618→  async function handleSubmit() {
   619→    if (!validLength || !allDigits || !match) return;
   620→    setSubmitting(true);
   621→    try {
   622→      await onConfirm(pin);

   828→function BiometricRow({
   829→  enabled,
   830→  onToggle,
   831→}: {
   832→  enabled: boolean;
   833→  onToggle: (v: boolean) => void;
   834→}) {
   835→  const biometricList = useBiometricList();
   836→  const registerMut = useRegisterBiometric();
   837→  const deleteMut = useDeleteBiometric();
   838→  const [registering, setRegistering] = React.useState(false);
   839→
   840→  async function handleRegister() {
   841→    if (typeof window === "undefined" || !window.PublicKeyCredential) {
   842→      toast.error("Browser tidak mendukung WebAuthn.");
   843→      return;
   844→    }
   845→    try {
   846→      const available =
   847→        await PublicKeyCredential.isUserVerifyingPlatformAuthenticatorAvailable();
   848→      if (!available) {
   849→        toast.error("Tidak ada autentikator biometrik pada perangkat ini.");
   850→        return;
   851→      }
   852→    } catch {
   853→      toast.error("Gagal memeriksa dukungan biometrik.");
   854→      return;
   855→    }
   856→
   857→    setRegistering(true);
   858→    try {
   859→      const challenge = new Uint8Array(32);
   860→      crypto.getRandomValues(challenge);
   861→      const userId = new TextEncoder().encode(
   862→        `dompetku-user-${Date.now()}`,
   863→      );
   864→
   865→      const credential = (await navigator.credentials.create({
   866→        publicKey: {
   867→          challenge,
   868→          rp: { name: "DompetKu" },
   869→          user: {
   870→            id: userId,
   871→            name: "user@dompetku",
   872→            displayName: "DompetKu User",
   873→          },
   874→          pubKeyCredParams: [
   875→            { type: "public-key", alg: -7 },
   876→            { type: "public-key", alg: -257 },
   877→          ],
   878→          authenticatorSelection: {
   879→            authenticatorAttachment: "platform",
   880→            userVerification: "required",
   881→            residentKey: "preferred",
   882→          },
   883→          timeout: 60_000,
   884→          attestation: "none",
   885→        },
   886→      })) as PublicKeyCredential | null;
   887→
   888→      if (!credential) {
   889→        toast.error("Pendaftaran biometrik dibatalkan.");
   890→        return;
   891→      }
   892→
   893→      const credentialId = bufferToBase64(credential.rawId);
   894→      const today = new Date().toISOString().slice(0, 10);
   895→      const name = `Device Fingerprint ${today}`;
   896→
   897→      await registerMut.mutateAsync({
   898→        name,
   899→        credentialId,
   900→        publicKey: credentialId,
   901→        counter: 0,
   902→      });
   903→      auditLog(
   904→        AUDIT_ACTIONS.BIOMETRIC_REGISTER,
   905→        `Mendaftarkan biometrik: ${name}`,
   906→        true,
   907→      );
   908→      toast.success("Biometrik berhasil didaftarkan.");
   909→      if (!enabled) onToggle(true);
   910→    } catch (err) {
   911→      const msg = err instanceof Error ? err.message : String(err);
   912→      if (msg.includes("cancel") || msg.includes("Abort")) {
   913→        toast.error("Pendaftaran dibatalkan.");
   914→      } else {
   915→        toast.error("Gagal mendaftarkan biometrik.");
   916→      }
   917→    } finally {
   918→      setRegistering(false);
   919→    }
   920→  }
   921→
   922→  function handleDelete(id: string, name: string) {
   923→    deleteMut.mutate(id, {
   924→      onSuccess: () => {
   925→        auditLog(
   926→          AUDIT_ACTIONS.BIOMETRIC_REGISTER,
   927→          `Menghapus biometrik: ${name}`,
   928→          true,
   929→        );
   930→        toast.success("Biometrik dihapus.");
   931→      },
   932→      onError: () => toast.error("Gagal menghapus biometrik."),
   933→    });
   934→  }
   935→
   936→  const items = biometricList.data ?? [];
   937→
   938→  return (
   939→    <>
   940→      <SettingRow
   941→        icon={<Fingerprint className="h-4 w-4" />}
   942→        title="Biometrik"
   943→        description="Buka kunci dengan sidik jari / wajah via WebAuthn."
   944→      >
   945→        <Switch
   946→          checked={enabled}
   947→          onCheckedChange={onToggle}
   948→          aria-label="Aktifkan biometrik"
   949→        />
   950→      </SettingRow>
   951→
   952→      {enabled && (
   953→        <div className="py-3">
   954→          <div className="mb-2 flex items-center justify-between">
   955→            <p className="text-xs font-medium text-muted-foreground">
   956→              Kredensial Terdaftar
   957→            </p>
   958→            <Button
   959→              size="sm"
   960→              variant="outline"
   961→              onClick={handleRegister}
   962→              disabled={registering}
   963→              className="gap-1.5"
   964→            >
   965→              {registering ? (
   966→                <Loader2 className="h-3.5 w-3.5 animate-spin" />
   967→              ) : (
   968→                <Plus className="h-3.5 w-3.5" />
   969→              )}
   970→              Daftarkan
   971→            </Button>
   972→          </div>
   973→          {items.length === 0 ? (
   974→            <p className="rounded-md bg-muted/60 px-3 py-2 text-xs text-muted-foreground">
   975→              Belum ada kredensial biometrik terdaftar.
   976→            </p>
   977→          ) : (
   978→            <div className="max-h-40 space-y-1.5 overflow-y-auto custom-scrollbar pr-1">
   979→              {items.map((b) => (
   980→                <div
   981→                  key={b.id}
   982→                  className="flex items-center justify-between gap-2 rounded-md border border-border bg-card px-3 py-2"
   983→                >
   984→                  <div className="min-w-0">
   985→                    <p className="truncate text-xs font-medium">{b.name}</p>
   986→                    <p className="text-[11px] text-muted-foreground">
   987→                      {formatTimeAgo(b.createdAt)}
   988→                    </p>
   989→                  </div>
   990→                  <Button
   991→                    size="sm"
   992→                    variant="ghost"
   993→                    className="h-7 w-7 p-0 text-rose-500 hover:bg-rose-500/10 hover:text-rose-600"
   994→                    onClick={() => handleDelete(b.id, b.name)}
   995→                    disabled={deleteMut.isPending}
   996→                    aria-label={`Hapus ${b.name}`}
   997→                  >
   998→                    <Trash2 className="h-3.5 w-3.5" />
   999→                  </Button>
  1000→                </div>
  1001→              ))}
  1002→            </div>
  1003→          )}
  1004→        </div>
  1005→      )}
  1006→    </>
  1007→  );
  1008→}

  1369→function MultiSelectPopover({
  1370→  type,
  1371→  value,
  1372→  onChange,
  1373→}: {
  1374→  type: "category" | "account";
  1375→  value: string;
  1376→  onChange: (v: string) => void;
  1377→}) {
  1378→  const cats = useCategories();
  1379→  const accs = useAccounts();
  1380→
  1381→  const items = React.useMemo(() => {
  1382→    if (type === "category") {
  1383→      return (cats.data ?? []).map((c) => ({
  1384→        id: c.id,
  1385→        label: c.name,
  1386→        color: c.color,
  1387→      }));
  1388→    }
  1389→    return (accs.data ?? []).map((a) => ({
  1390→      id: a.id,
  1391→      label: a.name,
  1392→      color: a.color,
  1393→    }));
  1394→  }, [type, cats.data, accs.data]);
  1395→
  1396→  const selected = React.useMemo(
  1397→    () => (value ? value.split(",").filter(Boolean) : []),
  1398→    [value],
  1399→  );
  1400→
  1401→  function toggle(id: string) {
  1402→    const next = selected.includes(id)
  1403→      ? selected.filter((x) => x !== id)
  1404→      : [...selected, id];
  1405→    onChange(next.join(","));
  1406→  }
  1407→
  1408→  const triggerLabel =
  1409→    selected.length === 0
  1410→      ? "Pilih…"
  1411→      : `${selected.length} dipilih`;
  1412→
  1413→  return (
  1414→    <Popover>
  1415→      <PopoverTrigger asChild>
  1416→        <Button variant="outline" size="sm" className="gap-1.5">
  1417→          <Check className="h-3.5 w-3.5 text-primary" />
  1418→          {triggerLabel}
  1419→        </Button>
  1420→      </PopoverTrigger>
  1421→      <PopoverContent className="w-72 p-0" align="end">
  1422→        <ScrollArea className="h-64">
  1423→          <div className="space-y-0.5 p-2">
  1424→            {items.length === 0 && (
  1425→              <p className="px-3 py-4 text-center text-xs text-muted-foreground">
  1426→                Tidak ada item.
  1427→              </p>
  1428→            )}
  1429→            {items.map((item) => {
  1430→              const checked = selected.includes(item.id);
  1431→              return (
  1432→                <label
  1433→                  key={item.id}
  1434→                  className={cn(
  1435→                    "flex cursor-pointer items-center gap-2.5 rounded-md px-2.5 py-2 text-sm transition-colors hover:bg-muted",
  1436→                    checked && "bg-muted",
  1437→                  )}
  1438→                >
  1439→                  <Checkbox
  1440→                    checked={checked}
  1441→                    onCheckedChange={() => toggle(item.id)}
  1442→                  />
  1443→                  <span
  1444→                    className="h-2.5 w-2.5 rounded-full"
  1445→                    style={{ backgroundColor: item.color }}
  1446→                  />
  1447→                  <span className="flex-1 truncate">{item.label}</span>
  1448→                </label>
  1449→              );
  1450→            })}
  1451→          </div>
  1452→        </ScrollArea>
