    79→function getRoundUpGoalId(): string | null {
    80→  if (typeof window === "undefined") return null;
    81→  try {
    82→    return window.localStorage.getItem(ROUNDUP_KEY);
    83→  } catch {
    84→    return null;
    85→  }
    86→}
    87→
    88→function setRoundUpGoalId(id: string | null) {
    89→  if (typeof window === "undefined") return;
    90→  try {
    91→    if (id) window.localStorage.setItem(ROUNDUP_KEY, id);
    92→    else window.localStorage.removeItem(ROUNDUP_KEY);
    93→  } catch {
    94→    /* ignore */
    95→  }
    96→}
    97→
    98→function getProcessedTxIds(): Set<string> {
    99→  if (typeof window === "undefined") return new Set();
   100→  try {
   101→    const raw = window.localStorage.getItem(PROCESSED_TX_KEY);
   102→    if (!raw) return new Set();
   103→    const arr = JSON.parse(raw) as string[];
   104→    return new Set(arr);
   105→  } catch {
   106→    return new Set();
   107→  }
   108→}
   109→
   110→function markTxProcessed(id: string) {
   111→  if (typeof window === "undefined") return;
   112→  try {
   113→    const set = getProcessedTxIds();
   114→    if (set.has(id)) return;
   115→    set.add(id);
   116→    // Cap to last 500 ids
   117→    const arr = Array.from(set).slice(-500);
   118→    window.localStorage.setItem(PROCESSED_TX_KEY, JSON.stringify(arr));
   119→  } catch {
   120→    /* ignore */
   121→  }
   122→}
   123→
   124→/** Returns milestones (25/50/75/100) already seen (celebrated) for this goal. */
   125→function getSeenMilestones(goalId: string): Set<number> {
   126→  if (typeof window === "undefined") return new Set();
   127→  try {
   128→    const raw = window.localStorage.getItem(MILESTONE_SEEN_KEY);
   129→    if (!raw) return new Set();
   130→    const map = JSON.parse(raw) as Record<string, number[]>;
   131→    return new Set(map[goalId] ?? []);
   132→  } catch {
   133→    return new Set();
   134→  }
   135→}
   136→
   137→function markMilestoneSeen(goalId: string, milestone: number) {
   138→  if (typeof window === "undefined") return;
   139→  try {
   140→    const raw = window.localStorage.getItem(MILESTONE_SEEN_KEY);
   141→    const map: Record<string, number[]> = raw
   142→      ? (JSON.parse(raw) as Record<string, number[]>)
   143→      : {};
   144→    const arr = Array.from(new Set([...(map[goalId] ?? []), milestone]));
   145→    map[goalId] = arr;
   146→    window.localStorage.setItem(MILESTONE_SEEN_KEY, JSON.stringify(map));
   147→  } catch {
   148→    /* ignore */
   149→  }
   150→}

   152→/** Find fun-fact comparison for the given amount (closest match below). */
   153→function findFunFact(amount: number): string | null {
   154→  if (!Number.isFinite(amount) || amount <= 0) return null;
   155→  // pick the largest FUN_FACTS amount <= given, otherwise smallest
   156→  const sorted = [...FUN_FACTS].sort((a, b) => a.amount - b.amount);
   157→  let match = sorted[0];
   158→  for (const f of sorted) {
   159→    if (f.amount <= amount) match = f;
   160→  }
   161→  if (!match) return null;
   162→  return match.comparisons[0] ?? null;
   163→}

   165→const MILESTONES = [25, 50, 75, 100];

   167→export function GoalsSection() {
   168→  const { data: goals, isLoading } = useGoals();
   169→  const { data: transactions } = useTransactions();
   170→  const updateGoalMut = useUpdateGoal();
   171→  const [dialogOpen, setDialogOpen] = React.useState(false);
   172→  const [editGoal, setEditGoal] = React.useState<Goal | null>(null);
   173→  const [contributionGoal, setContributionGoal] =
   174→    React.useState<Goal | null>(null);
   175→  const [confettiActive, setConfettiActive] = React.useState(false);
   176→
   177→  // Round-up processor: for each new EXPENSE transaction, if a round-up goal is
   178→  // active and not yet processed, round amount up to nearest Rp1000 and add the
   179→  // difference to that goal.
   180→  React.useEffect(() => {
   181→    if (!transactions || transactions.length === 0) return;
   182→    const roundUpGoalId = getRoundUpGoalId();
   183→    if (!roundUpGoalId) return;
   184→    const targetGoal = (goals ?? []).find((g) => g.id === roundUpGoalId);
   185→    if (!targetGoal || targetGoal.completed) return;
   186→
   187→    const processed = getProcessedTxIds();
   188→    const newOnes = transactions.filter(
   189→      (t) =>
   190→        t.type === "EXPENSE" &&
   191→        !processed.has(t.id) &&
   192→        !t.id.startsWith("temp-") &&
   193→        !t.isRecurringGenerated
   194→    );
   195→    if (newOnes.length === 0) return;
   196→
   197→    let totalDiff = 0;
   198→    for (const t of newOnes) {
   199→      const rounded = Math.ceil(t.amount / 1000) * 1000;
   200→      const diff = rounded - t.amount;
   201→      if (diff > 0) totalDiff += diff;
   202→      markTxProcessed(t.id);
   203→    }
   204→    // Mark temp/skipped transactions as processed too so they don't reprocess
   205→    for (const t of transactions) {
   206→      if (t.id.startsWith("temp-") || t.isRecurringGenerated) {
   207→        markTxProcessed(t.id);
   208→      }
   209→    }
   210→    if (totalDiff <= 0) return;
   211→    const newAmount = targetGoal.currentAmount + totalDiff;
   212→    updateGoalMut.mutate(
   213→      { id: targetGoal.id, data: { currentAmount: newAmount } },
   214→      {
   215→        onSuccess: () => {
   216→          toast.success(
   217→            `Round-up +${formatCurrency(totalDiff)} masuk ke "${targetGoal.name}".`
   218→          );
   219→        },
   220→      }
   221→    );
   222→  }, [transactions, goals, updateGoalMut]);
   223→
   224→  // Milestone celebration: detect 25/50/75/100% crossings.
   225→  React.useEffect(() => {
   226→    if (!goals || goals.length === 0) return;
   227→    let triggered: { goal: Goal; milestone: number } | null = null;
   228→    for (const g of goals) {
   229→      if (g.targetAmount <= 0) continue;
   230→      const pct = (g.currentAmount / g.targetAmount) * 100;
   231→      const seen = getSeenMilestones(g.id);
   232→      for (const m of MILESTONES) {
   233→        if (pct >= m && !seen.has(m)) {
   234→          markMilestoneSeen(g.id, m);
   235→          triggered = { goal: g, milestone: m };
   236→          break;
   237→        }
   238→      }
   239→      if (triggered) break;
   240→    }
   241→    if (triggered) {
   242→      const { goal, milestone } = triggered;
   243→      toast.success(
   244→        `🎉 Target "${goal.name}" sudah ${milestone}%!`
   245→      );
   246→      setConfettiActive(true);
   247→      const t = setTimeout(() => setConfettiActive(false), 3500);
   248→      return () => clearTimeout(t);
   249→    }
   250→  }, [goals]);
   251→
   252→  function openCreate() {
   253→    setEditGoal(null);
   254→    setDialogOpen(true);
   255→  }
   256→
   257→  function openEdit(g: Goal) {
   258→    setEditGoal(g);
   259→    setDialogOpen(true);
   260→  }
   261→
   262→  function openContribution(g: Goal) {
   263→    setContributionGoal(g);
   264→  }
   265→
   266→  // Round-up state (which goal is currently active)
   267→  const [roundUpGoalId, setRoundUpGoalIdState] = React.useState<string | null>(
   268→    null
   269→  );
   270→  React.useEffect(() => {
   271→    setRoundUpGoalIdState(getRoundUpGoalId());
   272→  }, []);
   273→
   274→  function toggleRoundUp(goalId: string) {
   275→    const current = getRoundUpGoalId();
   276→    const next = current === goalId ? null : goalId;
   277→    setRoundUpGoalId(next);
   278→    setRoundUpGoalIdState(next);
   279→    if (next) {
   280→      const g = (goals ?? []).find((x) => x.id === next);
   281→      if (g) {
   282→        toast.success(
   283→          `Round-up aktif untuk "${g.name}". Setiap transaksi dibulatkan ke atas.`
   284→        );
   285→      }
   286→    } else {
   287→      toast.info("Round-up dimatikan.");
   288→    }
   289→  }
   290→
   291→  return (
   292→    <div className="space-y-6">
   293→      {/* Header */}
   294→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   295→        <div>
   296→          <h2 className="text-xl font-semibold tracking-tight">
   297→            Target Tabungan
   298→          </h2>
   299→          <p className="mt-0.5 text-sm text-muted-foreground">
   300→            Tetapkan tujuan finansial & lacak progresnya.
   301→          </p>
   302→        </div>
   303→        <Button onClick={openCreate} className="gap-1">
   304→          <Plus className="h-4 w-4" />
   305→          Tambah Target
   306→        </Button>
   307→      </div>
   308→
   309→      {/* Grid */}
   310→      {isLoading ? (
   311→        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   312→          {Array.from({ length: 3 }).map((_, i) => (
   313→            <Skeleton key={i} className="h-56 rounded-2xl" />
   314→          ))}
   315→        </div>
   316→      ) : (goals ?? []).length === 0 ? (
   317→        <EmptyState onCreate={openCreate} />
   318→      ) : (
   319→        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   320→          {(goals ?? []).map((g) => (
   321→            <GoalCard
   322→              key={g.id}
   323→              goal={g}
   324→              transactions={(transactions ?? []).filter(
   325→                (t) => t.goalId === g.id
   326→              )}
   327→              roundUpActive={roundUpGoalId === g.id}
   328→              onToggleRoundUp={() => toggleRoundUp(g.id)}
   329→              onEdit={() => openEdit(g)}
   330→              onContribute={() => openContribution(g)}
   331→            />
   332→          ))}
   333→        </div>
   334→      )}
   335→
   336→      <GoalFormDialog
   337→        open={dialogOpen}
   338→        onOpenChange={setDialogOpen}
   339→        editGoal={editGoal}
   340→      />
   341→
   342→      <ContributionDialog
   343→        goal={contributionGoal}
   344→        onOpenChange={(o) => !o && setContributionGoal(null)}
   345→      />
   346→
   347→      {confettiActive && <ConfettiOverlay />}
   348→    </div>
   349→  );
   350→}

   409→function ProgressRing({
   410→  percentage,
   411→  color,
   412→  size = 64,
   413→}: {
   414→  percentage: number;
   415→  color: string;
   416→  size?: number;
   417→}) {
   418→  const strokeWidth = 6;
   419→  const r = (size - strokeWidth * 2) / 2;
   420→  const c = 2 * Math.PI * r;
   421→  const clamped = Math.min(Math.max(percentage, 0), 100);
   422→  const offset = c - (clamped / 100) * c;
   423→  return (
   424→    <div
   425→      className="relative inline-flex shrink-0"
   426→      style={{ width: size, height: size }}
   427→    >
   428→      <svg
   429→        width={size}
   430→        height={size}
   431→        className="-rotate-90"
   432→        aria-hidden="true"
   433→      >
   434→        <circle
   435→          cx={size / 2}
   436→          cy={size / 2}
   437→          r={r}
   438→          fill="none"
   439→          stroke="var(--muted)"
   440→          strokeWidth={strokeWidth}
   441→        />
   442→        <circle
   443→          cx={size / 2}
   444→          cy={size / 2}
   445→          r={r}
   446→          fill="none"
   447→          stroke={color}
   448→          strokeWidth={strokeWidth}
   449→          strokeDasharray={c}
   450→          strokeDashoffset={offset}
   451→          strokeLinecap="round"
   452→          className="transition-all duration-500"
   453→        />
   454→      </svg>
   455→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   456→        <span className="text-sm font-bold tabular-nums text-foreground">
   457→          {Math.round(clamped)}%
   458→        </span>
   459→      </div>
   460→    </div>
   461→  );
   462→}

   464→function GoalCard({
   465→  goal,
   466→  transactions,
   467→  roundUpActive,
   468→  onToggleRoundUp,
   469→  onEdit,
   470→  onContribute,
   471→}: {
   472→  goal: Goal;
   473→  transactions: Array<{ id: string; date: string | Date }>;
   474→  roundUpActive: boolean;
   475→  onToggleRoundUp: () => void;
   476→  onEdit: () => void;
   477→  onContribute: () => void;
   478→}) {
   479→  const deleteMut = useDeleteGoal();
   480→  const percentage =
   481→    goal.targetAmount > 0
   482→      ? (goal.currentAmount / goal.targetAmount) * 100
   483→      : 0;
   484→  const remaining = Math.max(goal.targetAmount - goal.currentAmount, 0);
   485→
   486→  // Streak: consecutive days with a contribution (transaction) for this goal.
   487→  const streak = React.useMemo(() => {
   488→    if (!transactions || transactions.length === 0) return 0;
   489→    return calculateStreak(transactions.map((t) => t.date));
   490→  }, [transactions]);
   491→
   492→  const funFact = React.useMemo(() => {
   493→    if (remaining <= 0) return null;
   494→    return findFunFact(remaining);
   495→  }, [remaining]);
   496→
   497→  function handleDelete() {
   498→    deleteMut.mutate(goal.id, {
   499→      onSuccess: () =>
   500→        toast.success(`Target "${goal.name}" dihapus.`),
   501→      onError: (err) =>
   502→        toast.error(err.message || "Gagal menghapus target."),
   503→    });
   504→  }
   505→
   506→  return (
   507→    <Card className="group relative p-4 transition-shadow hover:shadow-md">
   508→      {/* Top row */}
   509→      <div className="flex items-start justify-between gap-3">
   510→        <div className="flex min-w-0 items-center gap-3">
   511→          <ProgressRing percentage={percentage} color={goal.color} />
   512→          <div className="min-w-0">
   513→            <div className="flex items-center gap-1.5">
   514→              <span
   515→                className="flex h-7 w-7 shrink-0 items-center justify-center rounded-lg"
   516→                style={{ backgroundColor: `${goal.color}1a` }}
   517→              >
   518→                <LucideIcon
   519→                  name={goal.icon}
   520→                  className="h-4 w-4"
   521→                  style={{ color: goal.color }}
   522→                />
   523→              </span>
   524→              <p className="truncate text-sm font-semibold text-foreground">
   525→                {goal.name}
   526→              </p>
   527→            </div>
   528→            <p className="mt-1 text-xs text-muted-foreground">
   529→              {goal.targetDate
   530→                ? `Target: ${formatDate(goal.targetDate)}`
   531→                : "Tanpa tanggal target"}
   532→            </p>
   533→            {streak > 0 && (
   534→              <Badge
   535→                variant="secondary"
   536→                className="mt-1.5 gap-0.5 border-transparent bg-orange-100 text-orange-700 dark:bg-orange-500/15 dark:text-orange-400"
   537→                title={`${streak} hari berturut-turut menabung`}
   538→              >
   539→                <Flame className="h-3 w-3" />
   540→                {streak} hari nabung berturut
   541→              </Badge>
   542→            )}
   543→          </div>
   544→        </div>
   545→
   546→        {goal.completed && (
   547→          <Badge className="border-transparent bg-emerald-100 text-emerald-700 dark:bg-emerald-500/15 dark:text-emerald-400">
   548→            <Check className="h-3 w-3" />
   549→            Selesai
   550→          </Badge>
   551→        )}
   552→      </div>
   553→
   554→      {/* Amounts */}
   555→      <div className="mt-4 space-y-2">
   556→        <div className="flex items-end justify-between gap-2">
   557→          <div>
   558→            <p className="text-[11px] text-muted-foreground">
   559→              Terkumpul
   560→            </p>
   561→            <p className="text-sm font-semibold tabular-nums text-income">
   562→              {formatCurrency(goal.currentAmount)}
   563→            </p>
   564→          </div>
   565→          <div className="text-right">
   566→            <p className="text-[11px] text-muted-foreground">Target</p>
   567→            <p className="text-sm font-semibold tabular-nums text-foreground">
   568→              {formatCurrency(goal.targetAmount)}
   569→            </p>
   570→          </div>
   571→        </div>
   572→
   573→        {/* Linear progress bar (in addition to ring) */}
   574→        <div
   575→          className="h-1.5 w-full overflow-hidden rounded-full bg-muted"
   576→          role="progressbar"
   577→          aria-valuemin={0}
   578→          aria-valuemax={100}
   579→          aria-valuenow={Math.round(percentage)}
   580→        >
   581→          <div
   582→            className="h-full rounded-full transition-all duration-500"
   583→            style={{
   584→              width: `${Math.min(percentage, 100)}%`,
   585→              backgroundColor: goal.color,
   586→            }}
   587→          />
   588→        </div>
   589→
   590→        <p className="text-[11px] text-muted-foreground">
   591→          {goal.completed
   592→            ? "Target tercapai. Kerja bagus!"
   593→            : `Sisa ${formatCurrencyCompact(remaining)} lagi`}
   594→        </p>
   595→
   596→        {/* Fun fact comparison */}
   597→        {funFact && !goal.completed && (
   598→          <p className="flex items-center gap-1.5 rounded-md bg-emerald-50 px-2 py-1 text-[11px] text-emerald-700 dark:bg-emerald-500/10 dark:text-emerald-400">
   599→            <Sparkles className="h-3 w-3 shrink-0" />
   600→            <span>
   601→              <span className="font-semibold">{formatCurrencyCompact(remaining)}</span>
