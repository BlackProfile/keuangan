    96→function splitCategoryMeta(value: string) {
    97→  return (
    98→    SPLIT_BILL_CATEGORIES.find((c) => c.value === value) ??
    99→    SPLIT_BILL_CATEGORIES[SPLIT_BILL_CATEGORIES.length - 1]
   100→  );
   101→}
   102→
   103→function isOverdue(dueDate: string | null, settled: boolean): boolean {
   104→  if (!dueDate || settled) return false;
   105→  const today = new Date();
   106→  today.setHours(0, 0, 0, 0);
   107→  const due = new Date(dueDate);
   108→  due.setHours(0, 0, 0, 0);
   109→  return due.getTime() < today.getTime();
   110→}

   557→          </AlertDialogContent>
   558→        </AlertDialog>
   559→      </div>

   562→}

   593→    setTitle("");
   594→    setTotalAmount("");
   595→    setPaidBy("");
   596→    setSplitType("EQUAL");
   597→    setCategory("MAKAN");
   598→    setNote("");
   599→    setParticipants([
   600→      { name: "", share: "" },
   601→      { name: "", share: "" },
   602→    ]);
   603→    setError(null);
   604→  }, [open]);
   605→
   606→  const totalNum = Number(totalAmount) || 0;
   607→
   608→  // Recompute EQUAL shares when total/participant-count changes
   609→  const computedParticipants = React.useMemo(() => {
   610→    if (splitType !== "EQUAL") return participants;
   611→    const validCount = participants.filter((p) => p.name.trim()).length;
   612→    const share = validCount > 0 ? Math.round(totalNum / validCount) : 0;
   613→    return participants.map((p) =>
   614→      p.name.trim() ? { ...p, share: String(share) } : p
   615→    );
   616→  }, [participants, splitType, totalNum]);
   617→
   618→  const shareSum = React.useMemo(() => {
   619→    return computedParticipants
   620→      .filter((p) => p.name.trim())
   621→      .reduce((s, p) => s + (Number(p.share) || 0), 0);
   622→  }, [computedParticipants]);
   623→
   624→  const shareMismatch =
   625→    splitType === "CUSTOM" && totalNum > 0 && shareSum !== totalNum;
   626→
   627→  function addParticipant() {
   628→    setParticipants((prev) => [...prev, { name: "", share: "" }]);
   629→  }
   630→  function removeParticipant(idx: number) {
   631→    setParticipants((prev) => prev.filter((_, i) => i !== idx));
   632→  }
   633→  function updateParticipant(idx: number, field: "name" | "share", value: string) {
   634→    setParticipants((prev) =>
   635→      prev.map((p, i) => (i === idx ? { ...p, [field]: value } : p))
   636→    );
   637→  }
   638→
   639→  function handleSubmit(e: React.FormEvent) {
   640→    e.preventDefault();
   641→    setError(null);
   642→
   643→    if (!title.trim()) {
   644→      setError("Judul patungan wajib diisi.");
   645→      return;
   646→    }
   647→    if (!Number.isFinite(totalNum) || totalNum <= 0) {
   648→      setError("Total jumlah harus lebih dari 0.");
   649→      return;
   650→    }
   651→    if (!paidBy.trim()) {
   652→      setError("Isi siapa yang membayar.");
   653→      return;
   654→    }
   655→    const valid = computedParticipants.filter((p) => p.name.trim());
   656→    if (valid.length < 2) {
   657→      setError("Minimal 2 peserta untuk patungan.");
   658→      return;
   659→    }
   660→    if (shareMismatch) {
   661→      setError(
   662→        `Total share (${formatCurrency(shareSum)}) tidak sama dengan total (${formatCurrency(
   663→          totalNum
   664→        )}). Sesuaikan dulu.`
   665→      );
   666→      return;
   667→    }
   668→
   669→    const catMeta = splitCategoryMeta(category);
   670→    const payload: SplitBillInput = {
   671→      title: title.trim(),
   672→      description: undefined,
   673→      totalAmount: Math.round(totalNum),
   674→      paidBy: paidBy.trim(),
   675→      splitType,
   676→      category,
   677→      icon: catMeta.icon,
   678→      color: catMeta.color,
   679→      note: note.trim() || undefined,
   680→      participants: valid.map((p) => ({
   681→        name: p.name.trim(),
   682→        share: splitType === "EQUAL" ? Math.round(totalNum / valid.length) : Math.round(Number(p.share) || 0),
   683→      })),
   684→    };
   685→
   686→    createMut.mutate(payload, {
   687→      onSuccess: () => {
   688→        toast.success("Patungan dibuat.");
   689→        onOpenChange(false);
   690→      },
   691→      onError: (err) =>
   692→        setError(err.message || "Gagal membuat patungan."),
   693→    });
   694→  }
   695→
   696→  const pending = createMut.isPending;
   697→
   698→  return (
   699→    <Dialog open={open} onOpenChange={onOpenChange}>
   700→      <DialogContent
   701→        showCloseButton={false}
   702→        className="max-w-md gap-0 overflow-hidden p-0 sm:rounded-2xl"
   703→      >
   704→        <DialogHeader className="border-b border-border bg-muted/30 p-5 pb-4">
   705→          <div className="flex items-center justify-between">
   706→            <div>
   707→              <DialogTitle className="text-base">Buat Patungan</DialogTitle>
   708→              <DialogDescription className="text-xs">
   709→                Bagi tagihan bareng teman, otomatis terhitung share-nya.
   710→              </DialogDescription>
   711→            </div>
   712→            <DialogClose asChild>
   713→              <Button variant="ghost" size="icon" className="h-8 w-8 shrink-0">
   714→                <X className="h-4 w-4" />
   715→              </Button>
   716→            </DialogClose>
   717→          </div>
   718→        </DialogHeader>
   719→
   720→        <form onSubmit={handleSubmit} className="flex flex-col">
   721→          <div className="max-h-[70vh] space-y-4 overflow-y-auto p-5">
   722→            <div className="space-y-1.5">
   723→              <Label htmlFor="sb-title">Judul</Label>
   724→              <Input
   725→                id="sb-title"
   726→                placeholder="cth. Makan Ayam Geprek"
   727→                value={title}
   728→                onChange={(e) => setTitle(e.target.value)}
   729→                autoFocus
   730→                maxLength={60}
   731→              />
   732→            </div>
   733→
   734→            <div className="grid grid-cols-2 gap-3">
   735→              <div className="space-y-1.5">
   736→                <Label htmlFor="sb-total">Total (Rp)</Label>
   737→                <Input
   738→                  id="sb-total"
   739→                  type="number"
   740→                  inputMode="numeric"
   741→                  min={0}
   742→                  step={500}
   743→                  placeholder="cth. 60000"
   744→                  value={totalAmount}
   745→                  onChange={(e) => setTotalAmount(e.target.value)}
   746→                />
   747→                {totalNum > 0 && (
   748→                  <p className="text-xs text-muted-foreground">
   749→                    {formatCurrency(totalNum)}
   750→                  </p>
   751→                )}
   752→              </div>
   753→              <div className="space-y-1.5">
   754→                <Label htmlFor="sb-paidby">Dibayar Oleh</Label>
   755→                <Input
   756→                  id="sb-paidby"
   757→                  placeholder="cth. Aku / Andi"
   758→                  value={paidBy}
   759→                  onChange={(e) => setPaidBy(e.target.value)}
   760→                  maxLength={40}
   761→                />
   762→              </div>
   763→            </div>
   764→
   765→            <div className="grid grid-cols-2 gap-3">
   766→              <div className="space-y-1.5">
   767→                <Label htmlFor="sb-split">Tipe Bagi</Label>
   768→                <Select
   769→                  value={splitType}
   770→                  onValueChange={(v) => setSplitType(v as SplitTypeValue)}
   771→                >
   772→                  <SelectTrigger id="sb-split" className="w-full">
   773→                    <SelectValue />
   774→                  </SelectTrigger>
   775→                  <SelectContent>
   776→                    <SelectItem value="EQUAL">Sama Rata</SelectItem>
   777→                    <SelectItem value="CUSTOM">Custom</SelectItem>
   778→                  </SelectContent>
   779→                </Select>
   780→              </div>
   781→              <div className="space-y-1.5">
   782→                <Label htmlFor="sb-cat">Kategori</Label>
   783→                <Select
   784→                  value={category}
   785→                  onValueChange={(v) => setCategory(v as SplitCategoryValue)}
   786→                >
   787→                  <SelectTrigger id="sb-cat" className="w-full">
   788→                    <SelectValue />
   789→                  </SelectTrigger>
   790→                  <SelectContent>
   791→                    {SPLIT_BILL_CATEGORIES.map((c) => (
   792→                      <SelectItem key={c.value} value={c.value}>
   793→                        <span className="flex items-center gap-2">
   794→                          <LucideIcon
   795→                            name={c.icon}
   796→                            className="h-4 w-4"
   797→                            style={{ color: c.color }}
   798→                          />
   799→                          {c.label}
   800→                        </span>
   801→                      </SelectItem>
   802→                    ))}
   803→                  </SelectContent>
   804→                </Select>
   805→              </div>
   806→            </div>
   807→
   808→            {/* Participants */}
   809→            <div className="space-y-2">
   810→              <div className="flex items-center justify-between">
   811→                <Label>Peserta</Label>
   812→                <Button
   813→                  type="button"
   814→                  variant="outline"
   815→                  size="sm"
   816→                  onClick={addParticipant}
   817→                  className="h-7 gap-1 px-2 text-xs"
   818→                >
   819→                  <Plus className="h-3 w-3" />
   820→                  Tambah
   821→                </Button>
   822→              </div>
   823→
   824→              <div className="space-y-2">
   825→                {computedParticipants.map((p, idx) => (
   826→                  <div key={idx} className="flex items-center gap-2">
   827→                    <Input
   828→                      placeholder={`Nama peserta ${idx + 1}`}
   829→                      value={p.name}
   830→                      onChange={(e) =>
   831→                        updateParticipant(idx, "name", e.target.value)
   832→                      }
   833→                      maxLength={40}
   834→                    />
   835→                    <Input
   836→                      type="number"
   837→                      inputMode="numeric"
   838→                      placeholder="share"
   839→                      value={p.share}
   840→                      onChange={(e) =>
   841→                        updateParticipant(idx, "share", e.target.value)
   842→                      }
   843→                      disabled={splitType === "EQUAL"}
   844→                      className="w-28 tabular-nums"
   845→                    />
   846→                    {participants.length > 2 && (
   847→                      <Button
   848→                        type="button"
   849→                        variant="ghost"
   850→                        size="icon"
   851→                        onClick={() => removeParticipant(idx)}
   852→                        className="h-9 w-9 shrink-0 text-muted-foreground hover:text-destructive"
   853→                        aria-label="Hapus peserta"
   854→                      >
   855→                        <X className="h-4 w-4" />
   856→                      </Button>
   857→                    )}
   858→                  </div>
   859→                ))}
   860→              </div>
   861→
   862→              {splitType === "EQUAL" && totalNum > 0 && (
   863→                <p className="text-xs text-muted-foreground">
   864→                  Share otomatis dihitung rata:{" "}
   865→                  <span className="font-medium tabular-nums">
   866→                    {formatCurrency(
   867→                      computedParticipants.filter((p) => p.name.trim()).length >
   868→                        0
   869→                        ? Math.round(
   870→                            totalNum /
   871→                              computedParticipants.filter((p) =>
   872→                                p.name.trim()
   873→                              ).length
   874→                          )
   875→                        : 0
   876→                    )}
   877→                  </span>{" "}
   878→                  / orang
   879→                </p>
   880→              )}
   881→              {shareMismatch && (
   882→                <p className="text-xs text-amber-600 dark:text-amber-400">
   883→                  Total share ({formatCurrency(shareSum)}) belum sama dengan{" "}
   884→                  {formatCurrency(totalNum)}.
   885→                </p>
   886→              )}
   887→            </div>
   888→
   889→            <div className="space-y-1.5">
   890→              <Label htmlFor="sb-note">Catatan (opsional)</Label>
   891→              <Textarea
   892→                id="sb-note"
   893→                placeholder="cth. Bayar di restoran depan kampus"
   894→                value={note}
   895→                onChange={(e) => setNote(e.target.value)}
   896→                maxLength={120}
   897→                className="min-h-16"
   898→              />
   899→            </div>
   900→
   901→            {error && (
   902→              <div className="rounded-md bg-destructive/10 px-3 py-2 text-sm text-destructive">
   903→                {error}
   904→              </div>
   905→            )}
   906→          </div>
   907→
   908→          <DialogFooter className="border-t border-border bg-muted/30 p-4">
   909→            <DialogClose asChild>
   910→              <Button type="button" variant="outline" disabled={pending}>
   911→                Batal
   912→              </Button>
   913→            </DialogClose>
   914→            <Button
   915→              type="submit"
   916→              disabled={pending}
   917→              className="gap-1 bg-emerald-600 hover:bg-emerald-700"
   918→            >
   919→              {pending && <Loader2 className="h-4 w-4 animate-spin" />}
   920→              <Plus className="h-4 w-4" />
   921→              Buat Patungan
   922→            </Button>
   923→          </DialogFooter>
   924→        </form>
   925→      </DialogContent>
   926→    </Dialog>
   927→  );
   928→}
