   545→function SummaryStat({
   546→  label,
   547→  value,
   548→  variant,
   549→  loading,
   550→}: {
   551→  label: string;
   552→  value: number;
   553→  variant: "income" | "expense";
   554→  loading?: boolean;
   555→}) {
   556→  const isIncome = variant === "income";
   557→  return (
   558→    <Card className="p-3 sm:p-4">
   559→      <p className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground sm:text-xs">
   560→        {label}
   561→      </p>
   562→      {loading ? (
   563→        <Skeleton className="mt-1.5 h-5 w-20" />
   564→      ) : (
   565→        <p
   566→          className={cn(
   567→            "mt-1 text-sm font-bold tabular-nums sm:text-base",
   568→            isIncome ? "text-income" : "text-expense"
   569→          )}
   570→        >
   571→          {value < 0 && !isIncome ? "−" : ""}
   572→          {value >= 0 && isIncome ? "+" : ""}
   573→          {formatCurrency(Math.abs(value))}
   574→        </p>
   575→      )}
   576→    </Card>
   577→  );
   578→}

   640→  const duplicateMut = useDuplicateTransaction();

   643→
   644→  function handleToggleHide() {
   645→    hideMut.mutate(transaction.id, {
   646→      onSuccess: () => {
   647→        toast.success(transaction.isHidden ? "Transaksi ditampilkan" : "Transaksi disembunyikan");
   648→      },
   649→      onError: () => toast.error("Gagal mengubah status transaksi"),
   650→    });
   651→  }
   652→
   653→  const isIncome = transaction.type === "INCOME";
   654→  const cat = transaction.category;
   655→
   656→  const moodOpt = transaction.mood
   657→    ? MOOD_OPTIONS.find((o) => o.value === transaction.mood)
   658→    : null;
   659→  const priorityOpt = transaction.priority
   660→    ? PRIORITY_OPTIONS.find((o) => o.value === transaction.priority)
   661→    : null;
   662→  const paymentOpt = transaction.paymentMethod
   663→    ? PAYMENT_METHOD_OPTIONS.find((o) => o.value === transaction.paymentMethod)
   664→    : null;
   665→
   666→  const tags = parseTags(transaction.tags);
   667→
   668→  const flags: Array<{ label: string; color: string }> = [];
   669→  if (transaction.isDebt) flags.push({ label: "Hutang", color: "#ef4444" });
   670→  if (transaction.isReimbursable)
   671→    flags.push({ label: "Reimbursable", color: "#0891b2" });
   672→  if (transaction.isSubscription)
   673→    flags.push({ label: "Langganan", color: "#a855f7" });
   674→  if (transaction.isBusinessExpense)
   675→    flags.push({ label: "Bisnis", color: "#f97316" });
   676→  if (transaction.isTaxDeductible)
   677→    flags.push({ label: "Pajak", color: "#6b7280" });
   678→  if (transaction.isSplit) flags.push({ label: "Split", color: "#14b8a6" });
   679→
   680→  const hasFlagsRow = flags.length > 0 || tags.length > 0 || !!transaction.photoUrl;
   681→
   682→  function handleDuplicate() {
   683→    duplicateMut.mutate(transaction.id, {
   684→      onSuccess: () => toast.success("Transaksi berhasil diduplikat."),
   685→      onError: (e) => toast.error(e.message ?? "Gagal menduplikat transaksi."),
   686→    });
   687→  }
   688→
   689→  function handleTogglePin() {
   690→    pinMut.mutate(transaction.id, {
   691→      onSuccess: (updated) =>
   692→        toast.success(
   693→          updated.isPinned
   694→            ? "Transaksi disematkan."
   695→            : "Sematan dilepas."
   696→        ),
   697→      onError: (e) =>
   698→        toast.error(e.message ?? "Gagal mengubah sematan transaksi."),
   699→    });
   700→  }
   701→
   702→  return (
   703→    <div
   704→      className={cn(
   705→        "group flex items-center gap-3 px-3 py-2 transition-colors hover:bg-muted/40 sm:px-4",
   706→        !isFirst && "border-t border-border/60",
   707→      )}
   708→    >
   709→      {/* Category icon (with optional pinned ring) */}
   710→      <span
   711→        className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl"
   712→        style={{ backgroundColor: cat ? `${cat.color}1a` : undefined }}
   713→      >
   714→        <LucideIcon
   715→          name={cat?.icon ?? "Circle"}
   716→          className="h-4.5 w-4.5"
   717→          style={{ color: cat?.color }}
   718→        />
   719→      </span>
   720→
   721→      {/* Main content */}
   722→      <div className="min-w-0 flex-1">
   723→        {/* Title row: description + mood emoji + priority dot + payment icon + pin */}
   724→        <div className="flex items-center gap-1.5">
   725→          <p className="truncate text-sm font-medium text-foreground">
   726→            {transaction.description}
   727→          </p>
   728→          {moodOpt && (
   729→            <span
   730→              className="shrink-0 text-xs"
   731→              title={`Mood: ${moodOpt.label}`}
   732→              aria-label={`Mood: ${moodOpt.label}`}
   733→            >
   734→              {moodOpt.emoji}
   735→            </span>
   736→          )}
   737→          {priorityOpt && (
   738→            <span
   739→              className="h-1.5 w-1.5 shrink-0 rounded-full"
   740→              style={{ backgroundColor: priorityOpt.color }}
   741→              title={`Prioritas: ${priorityOpt.label}`}
   742→              aria-label={`Prioritas: ${priorityOpt.label}`}
   743→            />
   744→          )}
   745→          {paymentOpt && (
   746→            <LucideIcon
   747→              name={paymentOpt.icon}
   748→              className="h-3 w-3 shrink-0 text-muted-foreground"
   749→              aria-label={`Pembayaran: ${paymentOpt.label}`}
   750→            />
   751→          )}
   752→          {transaction.isPinned && (
   753→            <Pin
   754→              className="h-3 w-3 shrink-0 fill-primary text-primary"
   755→              aria-label="Disematkan"
   756→            />
   757→          )}
   758→        </div>
   759→
   760→        {/* Category + date row */}
   761→        <p className="mt-0.5 truncate text-xs text-muted-foreground">
   762→          {cat?.name ?? "Tanpa kategori"}
   763→          <span className="mx-1 text-border">·</span>
   764→          {relativeDay(transaction.date)}
   765→        </p>
   766→
   767→        {/* Flags row */}
   768→        {hasFlagsRow && (
   769→          <div className="mt-1.5 flex flex-wrap items-center gap-1">
   770→            {flags.map((f) => (
   771→              <span
   772→                key={f.label}
   773→                className="inline-flex items-center rounded-full px-1.5 py-0.5 text-[10px] font-medium leading-none"
   774→                style={{
   775→                  color: f.color,
   776→                  backgroundColor: `${f.color}1a`,
   777→                }}
   778→              >
   779→                {f.label}
   780→              </span>
   781→            ))}
   782→            {tags.map((tag) => (
   783→              <span
   784→                key={tag}
   785→                className="inline-flex items-center rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-medium leading-none text-muted-foreground"
   786→              >
   787→                #{tag}
   788→              </span>
   789→            ))}
   790→            {transaction.photoUrl && (
   791→              <span
   792→                className="inline-flex items-center rounded-full bg-muted px-1.5 py-0.5 text-[10px] leading-none text-muted-foreground"
   793→                title="Memiliki foto"
   794→              >
   795→                <Camera className="h-3 w-3" />
   796→              </span>
   797→            )}
   798→          </div>
   799→        )}
   800→      </div>
   801→
   802→      {/* Right side: amount + actions */}
   803→      <div className="flex shrink-0 items-center gap-2">
   804→        <div className="text-right">
   805→          <div
   806→            className={cn(
   807→              "text-sm font-semibold tabular-nums",
   808→              isIncome ? "text-income" : "text-expense"
   809→            )}
   810→          >
   811→            {isIncome ? "+" : "−"}
   812→            {formatCurrency(transaction.amount)}
   813→          </div>
   814→        </div>
   815→        {/* Actions: combine edit + menu into single dropdown to avoid overlap */}
   816→        <DropdownMenu>
   817→          <DropdownMenuTrigger asChild>
   818→            <Button
   819→              variant="ghost"
   820→              size="icon"
   821→              className="h-8 w-8 shrink-0 text-muted-foreground opacity-60 transition-opacity hover:bg-muted hover:text-foreground focus-visible:opacity-100 group-hover:opacity-100"
   822→              aria-label="Aksi transaksi"
   823→            >
   824→              <MoreVertical className="h-4 w-4" />
   825→            </Button>
   826→          </DropdownMenuTrigger>
   827→          <DropdownMenuContent align="end" className="w-44">
   828→            <DropdownMenuItem
   829→              onClick={onEdit}
   830→              className="gap-2"
   831→            >
   832→              <Pencil className="h-3.5 w-3.5" />
   833→              Edit
   834→            </DropdownMenuItem>
   835→            <DropdownMenuItem
   836→              onClick={handleDuplicate}
   837→              disabled={duplicateMut.isPending}
   838→            >
   839→              {duplicateMut.isPending ? (
   840→                <Loader2 className="h-4 w-4 animate-spin" />
   841→              ) : (
   842→                <Copy className="h-4 w-4" />
   843→              )}
   844→              Duplikat
   845→            </DropdownMenuItem>
   846→            <DropdownMenuItem
   847→              onClick={handleTogglePin}
   848→              disabled={pinMut.isPending}
   849→            >
   850→              {pinMut.isPending ? (
   851→                <Loader2 className="h-4 w-4 animate-spin" />
   852→              ) : (
   853→                <Pin className="h-4 w-4" />
   854→              )}
   855→              {transaction.isPinned ? "Lepas Sematan" : "Sematkan"}
   856→            </DropdownMenuItem>
   857→            <DropdownMenuItem
   858→              onClick={handleToggleHide}
   859→              disabled={hideMut.isPending}
   860→            >
   861→              {hideMut.isPending ? (
   862→                <Loader2 className="h-4 w-4 animate-spin" />
   863→              ) : transaction.isHidden ? (
   864→                <Eye className="h-4 w-4" />
   865→              ) : (
   866→                <EyeOff className="h-4 w-4" />
   867→              )}
   868→              {transaction.isHidden ? "Tampilkan" : "Sembunyikan"}
   869→            </DropdownMenuItem>
   870→            <DropdownMenuSeparator />
   871→            <DropdownMenuItem onClick={() => toast("Detail transaksi")}>
   872→              <Eye className="h-4 w-4" />
   873→              Lihat Detail
   874→            </DropdownMenuItem>
   875→          </DropdownMenuContent>
   876→        </DropdownMenu>
   877→      </div>
   878→    </div>
   879→  );
   880→}

   902→function useDebouncedValue<T>(value: T, delay: number): T {
   903→  const [debounced, setDebounced] = React.useState(value);
   904→  React.useEffect(() => {
   905→    const t = setTimeout(() => setDebounced(value), delay);
   906→    return () => clearTimeout(t);
   907→  }, [value, delay]);
   908→  return debounced;
   909→}
