   141→   141→function ProgressRing({
   142→   142→  percentage,
   143→   143→  color,
   144→   144→  size = 64,
   145→   145→}: {
   146→   146→  percentage: number;
   147→   147→  color: string;
   148→   148→  size?: number;
   149→   149→}) {
   150→   150→  const strokeWidth = 6;
   151→   151→  const r = (size - strokeWidth * 2) / 2;
   152→   152→  const c = 2 * Math.PI * r;
   153→   153→  const clamped = Math.min(Math.max(percentage, 0), 100);
   154→   154→  const offset = c - (clamped / 100) * c;
   155→   155→  return (
   156→   156→    <div
   157→   157→      className="relative inline-flex shrink-0"
   158→   158→      style={{ width: size, height: size }}
   159→   159→    >
   160→   160→      <svg
   161→   161→        width={size}
   162→   162→        height={size}
   163→   163→        className="-rotate-90"
   164→   164→        aria-hidden="true"
   165→   165→      >
   166→   166→        <circle
   167→   167→          cx={size / 2}
   168→   168→          cy={size / 2}
   169→   169→          r={r}
   170→   170→          fill="none"
   171→   171→          stroke="var(--muted)"
   172→   172→          strokeWidth={strokeWidth}
   173→   173→        />
   174→   174→        <circle
   175→   175→          cx={size / 2}
   176→   176→          cy={size / 2}
   177→   177→          r={r}
   178→   178→          fill="none"
   179→   179→          stroke={color}
   180→   180→          strokeWidth={strokeWidth}
   181→   181→          strokeDasharray={c}
   182→   182→          strokeDashoffset={offset}
   183→   183→          strokeLinecap="round"
   184→   184→          className="transition-all duration-500"
   185→   185→        />
   186→   186→      </svg>
   187→   187→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   188→   188→        <span className="text-sm font-bold tabular-nums text-foreground">
   189→   189→          {Math.round(clamped)}%
   190→   190→        </span>
   191→   191→      </div>
   192→   192→    </div>
   193→   193→  );
   194→   194→}

   387→   387→function GoalFormDialog({
   388→   388→  open,
   389→   389→  onOpenChange,
   390→   390→  editGoal,
   391→   391→}: {
   392→   392→  open: boolean;
   393→   393→  onOpenChange: (open: boolean) => void;
   394→   394→  editGoal: Goal | null;
   395→   395→}) {
   396→   396→  const isEdit = !!editGoal;
   397→   397→  const createMut = useCreateGoal();
   398→   398→  const updateMut = useUpdateGoal();
   399→   399→
   400→   400→  const [name, setName] = React.useState("");
   401→   401→  const [targetAmount, setTargetAmount] = React.useState("");
   402→   402→  const [currentAmount, setCurrentAmount] = React.useState("");
   403→   403→  const [targetDate, setTargetDate] = React.useState<Date | undefined>(
   404→   404→    undefined
   405→   405→  );
   406→   406→  const [icon, setIcon] = React.useState(GOAL_ICONS[0]);
   407→   407→  const [color, setColor] = React.useState(GOAL_COLORS[0]);
   408→   408→  const [error, setError] = React.useState<string | null>(null);
   409→   409→
   410→   410→  React.useEffect(() => {
   411→   411→    if (open) {
   412→   412→      if (editGoal) {
   413→   413→        setName(editGoal.name);
   414→   414→        setTargetAmount(String(editGoal.targetAmount));
   415→   415→        setCurrentAmount(String(editGoal.currentAmount));
   416→   416→        setTargetDate(
   417→   417→          editGoal.targetDate ? parseDateLocal(editGoal.targetDate) : undefined
   418→   418→        );
   419→   419→        setIcon(editGoal.icon);
   420→   420→        setColor(editGoal.color);
   421→   421→      } else {
   422→   422→        setName("");
   423→   423→        setTargetAmount("");
   424→   424→        setCurrentAmount("");
   425→   425→        setTargetDate(undefined);
   426→   426→        setIcon(GOAL_ICONS[0]);
   427→   427→        setColor(GOAL_COLORS[0]);
   428→   428→      }
   429→   429→      setError(null);
   430→   430→    }
   431→   431→  }, [open, editGoal]);
   432→   432→
   433→   433→  function handleSubmit(e: React.FormEvent) {
   434→   434→    e.preventDefault();
   435→   435→    setError(null);
   436→   436→    if (!name.trim()) {
   437→   437→      setError("Nama target wajib diisi.");
   438→   438→      return;
   439→   439→    }
   440→   440→    const target = Number(targetAmount);
   441→   441→    if (!Number.isFinite(target) || target <= 0) {
   442→   442→      setError("Target jumlah harus lebih dari 0.");
   443→   443→      return;
   444→   444→    }
   445→   445→    const current = Number(currentAmount || 0);
   446→   446→    if (!Number.isFinite(current) || current < 0) {
   447→   447→      setError("Jumlah saat ini tidak valid.");
   448→   448→      return;
   449→   449→    }
   450→   450→
   451→   451→    const payload: GoalInput = {
   452→   452→      name: name.trim(),
   453→   453→      targetAmount: Math.round(target),
   454→   454→      currentAmount: Math.round(current),
   455→   455→      targetDate: targetDate ? formatDateInput(targetDate) : undefined,
   456→   456→      icon,
   457→   457→      color,
   458→   458→    };
   459→   459→
   460→   460→    if (isEdit && editGoal) {
   461→   461→      updateMut.mutate(
   462→   462→        { id: editGoal.id, data: payload },
   463→   463→        {
   464→   464→          onSuccess: () => {
   465→   465→            toast.success("Target diperbarui.");
   466→   466→            onOpenChange(false);
   467→   467→          },
   468→   468→          onError: (err) =>
   469→   469→            setError(err.message || "Gagal memperbarui target."),
   470→   470→        }
   471→   471→      );
   472→   472→    } else {
   473→   473→      createMut.mutate(payload, {
   474→   474→        onSuccess: () => {
   475→   475→          toast.success("Target ditambahkan.");
   476→   476→          onOpenChange(false);
   477→   477→        },
   478→   478→        onError: (err) =>
   479→   479→          setError(err.message || "Gagal menambahkan target."),
   480→   480→      });
   481→   481→    }
   482→   482→  }
   483→   483→
   484→   484→  const pending = createMut.isPending || updateMut.isPending;
   485→   485→
   486→   486→  return (
   487→   487→    <Dialog open={open} onOpenChange={onOpenChange}>
   488→   488→      <DialogContent
   489→   489→        showCloseButton={false}
   490→   490→        className="max-h-[90vh] max-w-md gap-0 overflow-hidden p-0 sm:rounded-2xl"
   491→   491→      >
   492→   492→        <DialogHeader className="border-b border-border bg-muted/30 p-5 pb-4">
   493→   493→          <div className="flex items-center justify-between">
   494→   494→            <div>
   495→   495→              <DialogTitle className="text-base">
   496→   496→                {isEdit ? "Ubah Target" : "Target Baru"}
   497→   497→              </DialogTitle>
   498→   498→              <DialogDescription className="text-xs">
   499→   499→                {isEdit
   500→   500→                  ? "Perbarui detail target tabungan ini."
   501→   501→                  : "Tetapkan target finansial baru untuk ditgejar."}
   502→   502→              </DialogDescription>
   503→   503→            </div>
   504→   504→            <DialogClose asChild>
   505→   505→              <Button
   506→   506→                variant="ghost"
   507→   507→                size="icon"
   508→   508→                className="h-8 w-8 shrink-0"
   509→   509→              >
   510→   510→                <X className="h-4 w-4" />
   511→   511→              </Button>
   512→   512→            </DialogClose>

   732→   732→function ContributionDialog({
   733→   733→  goal,

   736→   736→  goal: Goal | null;
   737→   737→  onOpenChange: (open: boolean) => void;

   739→   739→  const updateMut = useUpdateGoal();
   740→   740→  const [amount, setAmount] = React.useState("");
   741→   741→  const [error, setError] = React.useState<string | null>(null);
   742→   742→

   744→   744→    if (goal) {
