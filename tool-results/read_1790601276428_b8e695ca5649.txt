     2→

     4→import { motion } from "framer-motion";
     5→import {

     9→  CalendarRange,

    21→  Wallet,
    22→} from "lucide-react";
    23→

    34→  Sparkline,
    35→} from "@/components/finance/chart-widgets";

    39→  formatCurrency,
    40→  formatCurrencyCompact,
    41→  formatDateInput,
    42→  getGreeting,
    43→  getMonthKey,
    44→  parseDateLocal,
    45→  relativeDay,
    46→} from "@/lib/format";
    47→import {
    48→  useAnalytics,
    49→  useDashboard,
    50→  useTransactions,
    51→} from "@/lib/hooks";
    52→import type { BudgetStatus, Goal, Transaction } from "@/lib/types";
    53→
    54→interface Props {
    55→  onAdd: () => void;
    56→  onEdit: (t: Transaction) => void;
    57→  onViewAll: () => void;
    58→}
    59→
    60→export function DashboardTab({ onAdd, onEdit, onViewAll }: Props) {
    61→  // Avoid hydration mismatch — compute greeting on client only.
    62→  const [greeting, setGreeting] = React.useState<string>("");
    63→  React.useEffect(() => {
    64→    setGreeting(getGreeting());
    65→  }, []);
    66→
    67→  // Show/hide balance toggle (persisted to localStorage)
    68→  const [showBalance, setShowBalance] = React.useState(true);
    69→  React.useEffect(() => {
    70→    try {
    71→      const saved = localStorage.getItem("dompetku:showBalance");
    72→      if (saved !== null) setShowBalance(saved !== "false");
    73→    } catch {
    74→      // ignore
    75→    }
    76→  }, []);
    77→  const toggleBalance = React.useCallback(() => {
    78→    setShowBalance((prev) => {
    79→      const next = !prev;
    80→      try {
    81→        localStorage.setItem("dompetku:showBalance", String(next));
    82→      } catch {
    83→        // ignore
    84→      }
    85→      return next;
    86→    });
    87→  }, []);
    88→
    89→  const monthKey = getMonthKey(new Date());
    90→  const { data, isLoading } = useDashboard(monthKey);
    91→  const { data: analytics } = useAnalytics(monthKey);
    92→
    93→  const summary = data?.summary;
    94→  const recent = data?.recentTransactions ?? [];
    95→  const budgets = data?.budgetStatuses ?? [];
    96→  const goals = data?.goals ?? [];
    97→
    98→  // Fetch current month's transactions for Top 5 jajan + counter jajan today.
    99→  const today = React.useMemo(() => new Date(), []);
   100→  const monthStartStr = React.useMemo(() => {
   101→    const d = new Date(today.getFullYear(), today.getMonth(), 1);
   102→    return formatDateInput(d);
   103→  }, [today]);
   104→  const todayStr = React.useMemo(() => formatDateInput(today), [today]);
   105→
   106→  // For weekly summary trend + last 14-day avg jajan counter:
   107→  // fetch from min(monthStart, lastWeekMonday) to today.
   108→  const lastWeekMondayStr = React.useMemo(() => {
   109→    const dayIdx = (today.getDay() + 6) % 7; // 0 = Monday
   110→    const thisMonday = addDays(today, -dayIdx);
   111→    const lastMonday = addDays(thisMonday, -7);
   112→    return formatDateInput(lastMonday);
   113→  }, [today]);
   114→  const fetchFromStr = monthStartStr < lastWeekMondayStr ? monthStartStr : lastWeekMondayStr;
   115→
   116→  const { data: rangeTx, isLoading: rangeLoading } = useTransactions({
   117→    from: fetchFromStr,
   118→    to: todayStr,
   119→    limit: 500,
   120→  });
   121→
   122→  // ---- Derived: today's EXPENSE transactions ----
   123→  const todayExpenses = React.useMemo(() => {
   124→    if (!rangeTx) return [];
   125→    return rangeTx.filter((t) => {
   126→      if (t.type !== "EXPENSE") return false;
   127→      const d = parseDateLocal(t.date);
   128→      const td = parseDateLocal(todayStr);
   129→      return (
   130→        d.getFullYear() === td.getFullYear() &&
   131→        d.getMonth() === td.getMonth() &&
   132→        d.getDate() === td.getDate()
   133→      );
   134→    });
   135→  }, [rangeTx, todayStr]);
   136→
   137→  // ---- Derived: avg daily jajan count (last 14 days) ----
   138→  const avgDailyJajan = React.useMemo(() => {
   139→    if (!rangeTx) return 0;
   140→    const countsByDay = new Map<string, number>();
   141→    const start = addDays(parseDateLocal(todayStr), -13);
   142→    start.setHours(0, 0, 0, 0);
   143→    for (let i = 0; i < 14; i++) {
   144→      const d = addDays(start, i);
   145→      countsByDay.set(formatDateInput(d), 0);
   146→    }
   147→    for (const t of rangeTx) {
   148→      if (t.type !== "EXPENSE") continue;
   149→      const d = parseDateLocal(t.date);
   150→      if (d < start) continue;
   151→      const key = formatDateInput(d);
   152→      countsByDay.set(key, (countsByDay.get(key) ?? 0) + 1);
   153→    }
   154→    const total = Array.from(countsByDay.values()).reduce((a, b) => a + b, 0);
   155→    return total / 14;
   156→  }, [rangeTx, todayStr]);
   157→
   158→  // ---- Derived: Top 5 jajan favorit (this month) ----
   159→  const top5Jajan = React.useMemo(() => {
   160→    if (!rangeTx) return [] as Array<{ description: string; count: number; total: number }>;
   161→    const monthStart = parseDateLocal(monthStartStr);
   162→    const map = new Map<string, { count: number; total: number }>();
   163→    for (const t of rangeTx) {
   164→      if (t.type !== "EXPENSE") continue;
   165→      const d = parseDateLocal(t.date);
   166→      if (d < monthStart) continue;
   167→      const key = (t.description || "").trim().toLowerCase();
   168→      if (!key) continue;
   169→      const cur = map.get(key) ?? { count: 0, total: 0 };
   170→      cur.count += 1;
   171→      cur.total += t.amount;
   172→      map.set(key, cur);
   173→    }
   174→    return Array.from(map.entries())
   175→      .map(([description, v]) => ({
   176→        description,
   177→        count: v.count,
   178→        total: v.total,
   179→      }))
   180→      .sort((a, b) => b.count - a.count || b.total - a.total)
   181→      .slice(0, 5);
   182→  }, [rangeTx, monthStartStr]);
   183→
   184→  // ---- Derived: Weekly summary (this week vs last week) ----
   185→  const weeklySummary = React.useMemo(() => {
   186→    const empty = {
   187→      thisIncome: 0,
   188→      thisExpense: 0,
   189→      lastExpense: 0,
   190→      balance: 0,
   191→      trendPct: 0, // negative = hemat (good), positive = boros (bad)
   192→    };
   193→    if (!rangeTx) return empty;
   194→    const dayIdx = (today.getDay() + 6) % 7; // 0 = Monday
   195→    const thisMonday = addDays(today, -dayIdx);
   196→    thisMonday.setHours(0, 0, 0, 0);
   197→    const lastMonday = addDays(thisMonday, -7);
   198→    const lastSunday = addDays(thisMonday, -1);
   199→    lastSunday.setHours(23, 59, 59, 999);
   200→
   201→    let thisIncome = 0;
   202→    let thisExpense = 0;
   203→    let lastExpense = 0;
   204→
   205→    for (const t of rangeTx) {
   206→      const d = parseDateLocal(t.date);
   207→      if (d >= thisMonday) {
   208→        if (t.type === "INCOME") thisIncome += t.amount;
   209→        else thisExpense += t.amount;
   210→      } else if (d >= lastMonday && d <= lastSunday) {
   211→        if (t.type === "EXPENSE") lastExpense += t.amount;
   212→      }
   213→    }
   214→    const trendPct =
   215→      lastExpense > 0
   216→        ? ((thisExpense - lastExpense) / lastExpense) * 100
   217→        : thisExpense > 0
   218→          ? -100
   219→          : 0;
   220→    return {
   221→      thisIncome,
   222→      thisExpense,
   223→      lastExpense,
   224→      balance: thisIncome - thisExpense,
   225→      trendPct,
   226→    };
   227→  }, [rangeTx, today]);
   228→
   229→  // ---- Derived: 14-day sparkline data (expenses per day) ----
   230→  const sparklineData = React.useMemo(() => {
   231→    if (!rangeTx) return [] as number[];
   232→    const todayDate = parseDateLocal(todayStr);
   233→    const days: number[] = [];
   234→    for (let i = 13; i >= 0; i--) {
   235→      const d = addDays(todayDate, -i);
   236→      const dStr = formatDateInput(d);
   237→      const total = rangeTx
   238→        .filter((t) => {
   239→          if (t.type !== "EXPENSE") return false;
   240→          return t.date.startsWith(dStr);
   241→        })
   242→        .reduce((s, t) => s + t.amount, 0);
   243→      days.push(total);
   244→    }
   245→    return days;
   246→  }, [rangeTx, todayStr]);
   247→
   248→  // ---- Derived: weekly mini bar chart data (7 hari, Sen-Min) ----
   249→  const weeklyBarData = React.useMemo(() => {
   250→    if (!rangeTx) return [];
   251→    const todayDate = parseDateLocal(todayStr);
   252→    const labels = ["Sn", "Sl", "Rb", "Km", "Jm", "Sb", "Mg"];
   253→    const todayDay = (todayDate.getDay() + 6) % 7; // 0=Sen
   254→    const result: { label: string; value: number; isToday?: boolean }[] = [];
   255→    // start from last Monday
   256→    const monday = addDays(todayDate, -todayDay);
   257→    for (let i = 0; i < 7; i++) {
   258→      const d = addDays(monday, i);
   259→      const dStr = formatDateInput(d);
   260→      const total = rangeTx
   261→        .filter((t) => {
   262→          if (t.type !== "EXPENSE") return false;
   263→          return t.date.startsWith(dStr);
   264→        })
   265→        .reduce((s, t) => s + t.amount, 0);
   266→      result.push({
   267→        label: labels[i],
   268→        value: total,
   269→        isToday: i === todayDay,
   270→      });
   271→    }
   272→    return result;
   273→  }, [rangeTx, todayStr]);
   274→
   275→  // ---- Derived: comparison bulan lalu (from analytics) ----
   276→  const monthComparison = analytics?.monthComparison;
   277→
   278→  // Top budget (most at-risk)
   279→  const topBudget = React.useMemo(() => {
   280→    return budgets.slice().sort((a, b) => b.percentage - a.percentage)[0];
   281→  }, [budgets]);
   282→
   283→  const topGoal = React.useMemo(() => {
   284→    return goals
   285→      .slice()
   286→      .sort(
   287→        (a, b) =>
   288→          b.currentAmount / Math.max(b.targetAmount, 1) -
   289→          a.currentAmount / Math.max(a.targetAmount, 1),
   290→      )[0];
   291→  }, [goals]);
   292→
   293→  return (
   294→    <div className="space-y-4">
   295→      {/* Hero — glassmorphism card with sparkline + hide/show balance */}
   296→      <motion.div
   297→        initial={{ opacity: 0, y: 8 }}
   298→        animate={{ opacity: 1, y: 0 }}
   299→        transition={{ duration: 0.3 }}
   300→      >
   301→        <Card className="relative overflow-hidden border-0 p-5 text-white shadow-2xl shadow-emerald-900/20 gradient-hero sm:p-6">
   302→          {/* Decorative orbs */}
   303→          <div className="absolute -right-16 -top-16 h-48 w-48 rounded-full bg-white/10 blur-2xl" />
   304→          <div className="absolute -bottom-20 right-20 h-40 w-40 rounded-full bg-white/5 blur-xl" />
   305→          {/* Sparkline in background */}
   306→          {sparklineData.length > 0 && !isLoading && (
   307→            <div className="pointer-events-none absolute inset-x-0 bottom-0 h-32 opacity-70">
   308→              <Sparkline
   309→                data={sparklineData}
   310→                width={400}
   311→                height={100}
   312→                className="h-full w-full"
   313→                strokeClassName="stroke-white/80"
   314→                fillClassName="fill-white/15"
   315→                strokeWidth={2.5}
   316→              />
   317→            </div>
   318→          )}
   319→          <div className="relative">
   320→            <div className="flex items-start justify-between gap-3">
   321→              <div className="min-w-0">
   322→                <p className="text-xs font-medium text-white/70">
   323→                  {greeting ? `${greeting} 👋` : "Halo"}
   324→                </p>
   325→                <p className="mt-0.5 flex items-center gap-1.5 text-sm text-white/90">
   326→                  Total saldo
   327→                  <button
   328→                    onClick={toggleBalance}
   329→                    className="text-white/60 transition-colors hover:text-white"
   330→                    aria-label={showBalance ? "Sembunyikan saldo" : "Tampilkan saldo"}
   331→                  >
   332→                    {showBalance ? (
   333→                      <Eye className="h-3.5 w-3.5" />
   334→                    ) : (
   335→                      <EyeOff className="h-3.5 w-3.5" />
   336→                    )}
   337→                  </button>
   338→                </p>
   339→              </div>
   340→              <Button
   341→                onClick={onAdd}
   342→                size="sm"
   343→                className="shrink-0 border border-white/20 bg-white/15 text-white backdrop-blur-md transition-all hover:bg-white/25 active:scale-95"
   344→              >
   345→                <Plus className="h-4 w-4" />
   346→                <span className="hidden sm:inline">Tambah</span>
   347→              </Button>
   348→            </div>
   349→
   350→            <div className="mt-4">
   351→              {isLoading ? (
   352→                <p className="text-3xl font-bold tracking-tight sm:text-4xl">···</p>
   353→              ) : showBalance ? (
   354→                <p className="text-3xl font-bold tracking-tight tabular-nums sm:text-4xl">
   355→                  <span className="text-lg font-semibold text-white/70 sm:text-xl">Rp </span>
   356→                  <AnimatedNumber
   357→                    value={summary?.balance ?? 0}
   358→                    format={(n) => n.toLocaleString("id-ID")}
   359→                  />
   360→                </p>
   361→              ) : (
   362→                <p className="text-3xl font-bold tracking-tight tabular-nums sm:text-4xl">
   363→                  •••••••
   364→                </p>
   365→              )}
   366→            </div>
   367→
   368→            {/* Income / Expense quick stats — glass pills */}
   369→            <div className="mt-5 grid grid-cols-2 gap-2 sm:gap-3">
   370→              <div className="flex items-center gap-2 rounded-2xl border border-white/10 bg-white/10 px-3 py-2.5 backdrop-blur-md transition-colors hover:bg-white/15 sm:py-3">
   371→                <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-xl bg-emerald-400/25 sm:h-9 sm:w-9">
   372→                  <TrendingUp className="h-4 w-4 text-emerald-100 sm:h-4.5 sm:w-4.5" />
   373→                </span>
   374→                <div className="min-w-0">
   375→                  <p className="text-[10px] text-white/65 sm:text-[11px]">
   376→                    Pemasukan
   377→                  </p>
   378→                  <p className="truncate text-xs font-semibold text-emerald-50 sm:text-sm">
   379→                    {isLoading
   380→                      ? "—"
   381→                      : showBalance
   382→                        ? formatCurrencyCompact(summary?.monthIncome ?? 0)
   383→                        : "•••"}
   384→                  </p>
   385→                </div>
   386→              </div>
   387→              <div className="flex items-center gap-2 rounded-2xl border border-white/10 bg-white/10 px-3 py-2.5 backdrop-blur-md transition-colors hover:bg-white/15 sm:py-3">
   388→                <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-xl bg-rose-400/25 sm:h-9 sm:w-9">
   389→                  <TrendingDown className="h-4 w-4 text-rose-100 sm:h-4.5 sm:w-4.5" />
   390→                </span>
   391→                <div className="min-w-0">
   392→                  <p className="text-[10px] text-white/65 sm:text-[11px]">
   393→                    Pengeluaran
   394→                  </p>
   395→                  <p className="truncate text-xs font-semibold text-rose-50 sm:text-sm">
   396→                    {isLoading
   397→                      ? "—"
   398→                      : showBalance
   399→                        ? formatCurrencyCompact(summary?.monthExpense ?? 0)
   400→                        : "•••"}
   401→                  </p>
   402→                </div>
   403→              </div>
   404→            </div>
   405→          </div>
   406→        </Card>
   407→      </motion.div>
   408→
   409→      {/* Bento Grid — Sisa Bulan + Transaksi count + Quick stats */}
   410→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   411→        <Card className="col-span-2 p-4 sm:col-span-2">
   412→          <div className="flex items-center justify-between gap-3">
   413→            <div className="min-w-0">
   414→              <p className="text-xs text-muted-foreground">Sisa bulan ini</p>
   415→              <p className="mt-0.5 text-xl font-bold tabular-nums sm:text-2xl">
   416→                {isLoading
   417→                  ? "···"
   418→                  : showBalance
   419→                    ? formatCurrency(summary?.monthBalance ?? 0)
   420→                    : "••••••"}
   421→              </p>
   422→              <p className="mt-0.5 text-[10px] text-muted-foreground/70">
   423→                Pemasukan − pengeluaran bulan ini
   424→              </p>
   425→            </div>
   426→            <div className="flex shrink-0 flex-col items-end gap-0.5 rounded-xl bg-muted/50 px-3 py-2">
   427→              <p className="text-[10px] text-muted-foreground">Transaksi</p>
   428→              <p className="text-base font-bold tabular-nums">
   429→                {isLoading ? "—" : summary?.monthTransactionCount ?? 0}
   430→              </p>
   431→            </div>
   432→          </div>
   433→        </Card>
   434→        {/* Mini stat: avg/day */}
   435→        <Card className="p-3">
   436→          <div className="flex flex-col gap-1">
   437→            <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-orange-500/10">
   438→              <Flame className="h-3.5 w-3.5 text-orange-500" />
   439→            </span>
   440→            <p className="text-[10px] text-muted-foreground">Rata-rata/hari</p>
   441→            <p className="truncate text-sm font-bold tabular-nums">
   442→              {rangeLoading
   443→                ? "—"
   444→                : showBalance
   445→                  ? formatCurrencyCompact(
   446→                      Math.round(
   447→                        (summary?.monthExpense ?? 0) /
   448→                          Math.max(new Date().getDate(), 1),
   449→                      ),
   450→                    )
   451→                  : "•••"}
   452→            </p>
   453→          </div>
   454→        </Card>
   455→        {/* Mini stat: savings rate */}
   456→        <Card className="p-3">
   457→          <div className="flex flex-col gap-1">
   458→            <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-primary/10">
   459→              <Sparkles className="h-3.5 w-3.5 text-primary" />
   460→            </span>
   461→            <p className="text-[10px] text-muted-foreground">Tingkat tabung</p>
   462→            <p className="truncate text-sm font-bold tabular-nums">
   463→              {isLoading
   464→                ? "—"
   465→                : `${Math.round((summary?.savingsRate ?? 0) * 100)}%`}
   466→            </p>
   467→          </div>
   468→        </Card>
   469→      </div>
   470→
   471→      {/* Counter Jajan Harian + Weekly Summary — 2 col grid */}
   472→      {todayExpenses.length > 0 && (
   473→        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   474→          <CounterJajanCard
   475→            count={todayExpenses.length}
   476→            avg={avgDailyJajan}
   477→            isLoading={rangeLoading}
   478→          />
   479→          <WeeklySummaryCard
   480→            summary={weeklySummary}
   481→            barData={weeklyBarData}
   482→            isLoading={rangeLoading}
   483→          />
   484→        </div>
   485→      )}
   486→      {todayExpenses.length === 0 && (
   487→        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   488→          <WeeklySummaryCard
   489→            summary={weeklySummary}
   490→            barData={weeklyBarData}
   491→            isLoading={rangeLoading}
   492→          />
   493→        </div>
   494→      )}
   495→
   496→      {/* Comparison Bulan Lalu + Top 5 Jajan Favorit — 2 col grid */}
   497→      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   498→        <ComparisonBulanLaluCard
   499→          current={monthComparison?.current.expense ?? 0}
   500→          previous={monthComparison?.previous.expense ?? 0}
   501→          changePct={monthComparison?.expenseChange ?? 0}
   502→          isLoading={isLoading}
   503→        />
   504→        <Top5JajanCard items={top5Jajan} isLoading={rangeLoading} />
   505→      </div>
   506→
   507→      {/* Budget + Goals — conditional, grid layout on sm+ */}
   508→      {(topBudget || topGoal) && (
   509→        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   510→          {topBudget && <BudgetMiniCard budget={topBudget} />}
   511→          {topGoal && <GoalMiniCard goal={topGoal} />}
   512→        </div>
   513→      )}
   514→
   515→      {/* Recent transactions */}
   516→      <Card className="p-4">
   517→        <div className="mb-3 flex items-center justify-between">
   518→          <div>
   519→            <h3 className="text-sm font-semibold">Transaksi Terbaru</h3>
   520→            <p className="mt-0.5 text-xs text-muted-foreground">
   521→              6 transaksi terakhir Anda
   522→            </p>
   523→          </div>
   524→          <Button variant="ghost" size="sm" onClick={onViewAll} className="gap-1">
   525→            Lihat semua
   526→            <ArrowRight className="h-3.5 w-3.5" />
   527→          </Button>
   528→        </div>
   529→
   530→        {isLoading ? (
   531→          <div className="space-y-2">
   532→            {Array.from({ length: 4 }).map((_, i) => (
   533→              <Skeleton key={i} className="h-11 w-full rounded-xl" />
   534→            ))}
   535→          </div>
   536→        ) : recent.length === 0 ? (
   537→          <EmptyTransactions onAdd={onAdd} />
   538→        ) : (
