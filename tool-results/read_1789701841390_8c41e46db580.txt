   203→   203→function KeamananSection({ settings }: { settings: AppSettings }) {
   204→   204→  const updateMut = useUpdateSetting();
   205→   205→  const [pinDialogOpen, setPinDialogOpen] = React.useState(false);
   206→   206→  const [pin, setPin] = React.useState("");
   207→   207→
   208→   208→  function handleSwitchChange(enabled: boolean) {
   209→   209→    if (enabled) {
   210→   210→      setPin("");
   211→   211→      setPinDialogOpen(true);
   212→   212→    } else {
   213→   213→      // Disable: clear pinHash and pinEnabled
   214→   214→      updateMut.mutate({ key: "pinHash", value: "" });
   215→   215→      updateMut.mutate({ key: "pinEnabled", value: "false" });
   216→   216→      toast.success("PIN dinonaktifkan.");
   217→   217→    }
   218→   218→  }
   219→   219→
   220→   220→  async function handleSavePin() {
   221→   221→    if (pin.length !== 4) {
   222→   222→      toast.error("PIN harus 4 digit.");
   223→   223→      return;
   224→   224→    }
   225→   225→    try {
   226→   226→      const hash = await hashPin(pin);
   227→   227→      updateMut.mutate({ key: "pinHash", value: hash });
   228→   228→      updateMut.mutate({ key: "pinEnabled", value: "true" });
   229→   229→      setPin("");
   230→   230→      setPinDialogOpen(false);
   231→   231→      toast.success("PIN aktif. Aplikasi akan terkunci saat dibuka.");
   232→   232→    } catch {
   233→   233→      toast.error("Gagal menghash PIN. Coba lagi.");
   234→   234→    }
   235→   235→  }
   236→   236→
   237→   237→  function toggleHideAmounts(enabled: boolean) {
   238→   238→    updateMut.mutate({
   239→   239→      key: "hideAmounts",
   240→   240→      value: enabled ? "true" : "false",
   241→   241→    });
   242→   242→    toast.success(
   243→   243→      enabled ? "Nominal disembunyikan." : "Nominal ditampilkan.",
   244→   244→    );
   245→   245→  }
   246→   246→
   247→   247→  return (
   248→   248→    <SectionCard
   249→   249→      icon={<ShieldCheck className="h-4 w-4" />}
   250→   250→      title="Keamanan"
   251→   251→      description="Lindungi privasi data keuangan Anda."
   252→   252→    >
   253→   253→      <SettingRow
   254→   254→        icon={<Lock className="h-4 w-4" />}
   255→   255→        title="Kunci PIN"
   256→   256→        description="Minta PIN 4 digit saat membuka aplikasi."
   257→   257→      >
   258→   258→        <Switch
   259→   259→          checked={settings.pinEnabled}
   260→   260→          onCheckedChange={handleSwitchChange}
   261→   261→          disabled={updateMut.isPending}
   262→   262→          aria-label="Aktifkan kunci PIN"
   263→   263→        />
   264→   264→      </SettingRow>
   265→   265→
   266→   266→      <SettingRow
   267→   267→        icon={<EyeOff className="h-4 w-4" />}
   268→   268→        title="Sembunyikan Nominal"
   269→   269→        description="Sembunyikan jumlah uang di semua layar."
   270→   270→      >
   271→   271→        <Switch
   272→   272→          checked={settings.hideAmounts}
   273→   273→          onCheckedChange={toggleHideAmounts}
   274→   274→          disabled={updateMut.isPending}
   275→   275→          aria-label="Sembunyikan nominal"
   276→   276→        />
   277→   277→      </SettingRow>
   278→   278→
   279→   279→      <Dialog open={pinDialogOpen} onOpenChange={setPinDialogOpen}>
   280→   280→        <DialogContent showCloseButton={false} className="max-w-sm">
   281→   281→          <DialogHeader>
   282→   282→            <DialogTitle>Buat PIN 4 Digit</DialogTitle>
   283→   283→            <DialogDescription>
   284→   284→              PIN ini akan diminta setiap kali aplikasi dibuka.
   285→   285→            </DialogDescription>
   286→   286→          </DialogHeader>
   287→   287→          <div className="flex flex-col items-center gap-4 py-4">
   288→   288→            <InputOTP
   289→   289→              maxLength={4}
   290→   290→              value={pin}
   291→   291→              onChange={(v) => setPin(v)}
   292→   292→              autoFocus
   293→   293→            >
   294→   294→              <InputOTPGroup>
   295→   295→                <InputOTPSlot index={0} />
   296→   296→                <InputOTPSlot index={1} />
   297→   297→                <InputOTPSlot index={2} />
   298→   298→                <InputOTPSlot index={3} />
   299→   299→              </InputOTPGroup>
   300→   300→            </InputOTP>
   301→   301→            <p className="text-center text-xs text-muted-foreground">
   302→   302→              Ingat PIN ini — tidak bisa dipulihkan jika lupa.
   303→   303→            </p>
   304→   304→          </div>
   305→   305→          <DialogFooter>
   306→   306→            <DialogClose asChild>
   307→   307→              <Button variant="outline">Batal</Button>
   308→   308→            </DialogClose>
   309→   309→            <Button
   310→   310→              onClick={handleSavePin}
   311→   311→              disabled={pin.length !== 4 || updateMut.isPending}
   312→   312→              className="gap-1.5"
   313→   313→            >
   314→   314→              {updateMut.isPending && (
   315→   315→                <Loader2 className="h-4 w-4 animate-spin" />
   316→   316→              )}
   317→   317→              Simpan PIN
   318→   318→            </Button>
   319→   319→          </DialogFooter>
   320→   320→        </DialogContent>
   321→   321→      </Dialog>
   322→   322→    </SectionCard>
   323→   323→  );
   324→   324→}

   390→   390→function PengingatSection({ settings }: { settings: AppSettings }) {
   391→   391→  const updateMut = useUpdateSetting();
   392→   392→  const [hour, setHour] = React.useState(String(settings.reminderHour));
   393→   393→
   394→   394→  React.useEffect(() => {
   395→   395→    setHour(String(settings.reminderHour));
   396→   396→  }, [settings.reminderHour]);
   397→   397→
   398→   398→  // Schedule in-app browser notifications while the app is open
   399→   399→  React.useEffect(() => {
   400→   400→    if (!settings.reminderEnabled) return;
   401→   401→    if (typeof window === "undefined" || !("Notification" in window)) return;
   402→   402→
   403→   403→    const targetHour = settings.reminderHour;
   404→   404→    const interval = setInterval(() => {
   405→   405→      const now = new Date();
   406→   406→      if (now.getHours() !== targetHour) return;
   407→   407→      const nowMin = now.getMinutes();
   408→   408→      // Fire once near the top of the hour (first 5 minutes)
   409→   409→      if (nowMin > 5) return;
   410→   410→      const key = `dompetku-reminder-${now.toDateString()}`;
   411→   411→      try {
   412→   412→        if (localStorage.getItem(key)) return;
   413→   413→        localStorage.setItem(key, "1");
   414→   414→        if (Notification.permission === "granted") {
   415→   415→          new Notification("DompetKu", {
   416→   416→            body: "Saatnya mencatat transaksi hari ini! 📒",
   417→   417→          });
   418→   418→        }
   419→   419→      } catch {
   420→   420→        // localStorage may be unavailable (private mode) — ignore
   421→   421→      }
   422→   422→    }, 60_000);
   423→   423→
   424→   424→    return () => clearInterval(interval);
   425→   425→  }, [settings.reminderEnabled, settings.reminderHour]);
   426→   426→
   427→   427→  async function handleToggleReminder(enabled: boolean) {
   428→   428→    if (enabled && typeof window !== "undefined" && "Notification" in window) {
   429→   429→      const perm = await Notification.requestPermission();
   430→   430→      if (perm !== "granted") {
   431→   431→        toast.warning(
   432→   432→          "Izin notifikasi ditolak. Pengingat aktif tapi tanpa notifikasi.",
   433→   433→        );
   434→   434→      }
   435→   435→    }
   436→   436→    updateMut.mutate({
   437→   437→      key: "reminderEnabled",
   438→   438→      value: enabled ? "true" : "false",
   439→   439→    });
   440→   440→    toast.success(enabled ? "Pengingat aktif." : "Pengingat dinonaktifkan.");
   441→   441→  }
   442→   442→
   443→   443→  function handleHourChange(v: string) {
   444→   444→    setHour(v);
   445→   445→    const num = Number(v);
   446→   446→    if (Number.isInteger(num) && num >= 0 && num <= 23) {
   447→   447→      updateMut.mutate({ key: "reminderHour", value: String(num) });
   448→   448→    }
   449→   449→  }
   450→   450→
   451→   451→  function handleTestNotification() {
   452→   452→    if (typeof window === "undefined" || !("Notification" in window)) {
   453→   453→      toast.error("Browser tidak mendukung notifikasi.");
   454→   454→      return;
   455→   455→    }
   456→   456→    if (Notification.permission !== "granted") {
   457→   457→      toast.warning("Berikan izin notifikasi dulu.");
   458→   458→      return;
   459→   459→    }
   460→   460→    new Notification("DompetKu", {
   461→   461→      body: "Ini contoh pengingat. 📒",
   462→   462→    });
   463→   463→    toast.success("Notifikasi tes dikirim.");
   464→   464→  }
   465→   465→
   466→   466→  return (
   467→   467→    <SectionCard
   468→   468→      icon={<Bell className="h-4 w-4" />}
   469→   469→      title="Pengingat"
   470→   470→      description="Pengingat harian untuk mencatat transaksi."
   471→   471→    >
   472→   472→      <SettingRow
   473→   473→        icon={<Clock className="h-4 w-4" />}
   474→   474→        title="Pengingat Catat Harian"
   475→   475→        description="Aktifkan pengingat untuk mencatat setiap hari."
   476→   476→      >
   477→   477→        <Switch
   478→   478→          checked={settings.reminderEnabled}
   479→   479→          onCheckedChange={handleToggleReminder}
   480→   480→          disabled={updateMut.isPending}
   481→   481→          aria-label="Aktifkan pengingat"
   482→   482→        />
   483→   483→      </SettingRow>
   484→   484→
   485→   485→      {settings.reminderEnabled && (
   486→   486→        <>
   487→   487→          <SettingRow
   488→   488→            title="Waktu Pengingat"
   489→   489→            description="Pengingat akan muncul setiap hari pada jam ini."
   490→   490→          >
   491→   491→            <div className="flex items-center gap-2">
   492→   492→              <Input
   493→   493→                type="number"
   494→   494→                min={0}
   495→   495→                max={23}
   496→   496→                value={hour}
   497→   497→                onChange={(e) => handleHourChange(e.target.value)}
   498→   498→                className="h-9 w-16 text-center"
   499→   499→                aria-label="Jam pengingat"
   500→   500→              />
   501→   501→              <span className="text-xs text-muted-foreground">:00</span>
   502→   502→            </div>
   503→   503→          </SettingRow>
   504→   504→
   505→   505→          <SettingRow
   506→   506→            title="Uji Notifikasi"
   507→   507→            description="Kirim notifikasi tes untuk memastikan izin aktif."
   508→   508→          >
   509→   509→            <Button
   510→   510→              variant="outline"
   511→   511→              size="sm"
   512→   512→              onClick={handleTestNotification}
   513→   513→              className="gap-1.5"
   514→   514→            >
   515→   515→              <Send className="h-3.5 w-3.5" />
   516→   516→              Tes
   517→   517→            </Button>
   518→   518→          </SettingRow>

   796→   796→// CSV parser (handles quoted fields with commas)
   797→   797→// ============================================================
   798→   798→
   799→   799→function parseCsv(text: string): {
   800→   800→  headers: string[];
   801→   801→  rows: Array<Record<string, string>>;
   802→   802→} {
   803→   803→  // Strip BOM if present
   804→   804→  const cleaned = text.replace(/^\uFEFF/, "");
   805→   805→  const lines = cleaned
   806→   806→    .split(/\r?\n/)
   807→   807→    .map((l) => l.trim())
   808→   808→    .filter((l) => l !== "");
   809→   809→  if (lines.length === 0) return { headers: [], rows: [] };
   810→   810→
   811→   811→  const splitLine = (line: string): string[] => {
   812→   812→    const out: string[] = [];
   813→   813→    let cur = "";
   814→   814→    let inQuotes = false;
   815→   815→    for (let i = 0; i < line.length; i++) {
   816→   816→      const ch = line[i];
   817→   817→      if (ch === '"') {
   818→   818→        if (inQuotes && line[i + 1] === '"') {
   819→   819→          cur += '"';
   820→   820→          i++;
   821→   821→        } else {
   822→   822→          inQuotes = !inQuotes;
   823→   823→        }
   824→   824→      } else if (ch === "," && !inQuotes) {
   825→   825→        out.push(cur);
   826→   826→        cur = "";
   827→   827→      } else {
   828→   828→        cur += ch;
   829→   829→      }
   830→   830→    }
   831→   831→    out.push(cur);
   832→   832→    return out.map((s) => s.trim());
   833→   833→  };
   834→   834→
   835→   835→  const headers = splitLine(lines[0]);
   836→   836→  const rows: Array<Record<string, string>> = [];
   837→   837→  for (let i = 1; i < lines.length; i++) {
   838→   838→    const cells = splitLine(lines[i]);
   839→   839→    const obj: Record<string, string> = {};
   840→   840→    headers.forEach((h, idx) => {
   841→   841→      obj[h] = cells[idx] ?? "";
   842→   842→    });
   843→   843→    rows.push(obj);
   844→   844→  }
   845→   845→  return { headers, rows };
   846→   846→}
