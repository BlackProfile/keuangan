    27→// Wrapper to broadcast entity changes (safe to call — no-op if socket not connected)
    28→function broadcastEntityChange(entity: string, action: string) {
    29→  try {
    30→    broadcastChange(entity, action);
    31→  } catch {
    32→    // ignore — realtime is best-effort
    33→  }
    34→}

    92→export function useCreateTransaction() {
    93→  const qc = useQueryClient();
    94→  return useMutation({
    95→    mutationFn: (data: TransactionInput) => api.createTransaction(data),
    96→    onMutate: async (data) => {
    97→      // Optimistic: add temp transaction to list cache
    98→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
    99→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   100→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   101→        if (!old) return old;
   102→        const temp = {
   103→          ...data,
   104→          id: `temp-${Date.now()}`,
   105→          createdAt: new Date().toISOString(),
   106→          updatedAt: new Date().toISOString(),
   107→          isRecurringGenerated: false,
   108→        } as Transaction;
   109→        return [temp, ...old];
   110→      });
   111→      return { prev };
   112→    },
   113→    onError: (_e, _data, ctx) => {
   114→      // Rollback on error
   115→      if (ctx?.prev) {
   116→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   117→      }
   118→    },
   119→    onSuccess: () => {
   120→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   121→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   122→      qc.invalidateQueries({ queryKey: queryKeys.analytics });
   123→      qc.invalidateQueries({ queryKey: queryKeys.budgetStatuses });
   124→      qc.invalidateQueries({ queryKey: queryKeys.accounts });
   125→      broadcastEntityChange("transaction", "create");
   126→    },
   127→  });
   128→}

   130→export function useUpdateTransaction() {
   131→  const qc = useQueryClient();
   132→  return useMutation({
   133→    mutationFn: ({ id, data }: { id: string; data: TransactionInput }) =>
   134→      api.updateTransaction(id, data),
   135→    onMutate: async ({ id, data }) => {
   136→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
   137→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   138→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   139→        if (!old) return old;
   140→        return old.map((t) => (t.id === id ? { ...t, ...data } as Transaction : t));
   141→      });
   142→      return { prev };
   143→    },
   144→    onError: (_e, _v, ctx) => {
   145→      if (ctx?.prev) {
   146→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   147→      }
   148→    },
   149→    onSuccess: () => {
   150→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   151→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   152→      qc.invalidateQueries({ queryKey: queryKeys.analytics });
   153→      qc.invalidateQueries({ queryKey: queryKeys.budgetStatuses });
   154→      qc.invalidateQueries({ queryKey: queryKeys.accounts });
   155→      broadcastEntityChange("transaction", "update");
   156→    },
   157→  });
   158→}

   160→export function useDeleteTransaction() {
   161→  const qc = useQueryClient();
   162→  return useMutation({
   163→    mutationFn: (id: string) => api.deleteTransaction(id),
   164→    onMutate: async (id) => {
   165→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
   166→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   167→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   168→        if (!old) return old;
   169→        return old.filter((t) => t.id !== id);
   170→      });
   171→      return { prev };
   172→    },
   173→    onError: (_e, _id, ctx) => {
   174→      if (ctx?.prev) {
   175→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   176→      }
   177→    },
   178→    onSuccess: () => {
   179→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   180→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   181→      qc.invalidateQueries({ queryKey: queryKeys.analytics });
   182→      qc.invalidateQueries({ queryKey: queryKeys.budgetStatuses });
   183→      qc.invalidateQueries({ queryKey: queryKeys.accounts });
   184→      broadcastEntityChange("transaction", "delete");
   185→    },
   186→  });

   806→export function useToggleHideTransaction() {
   807→  const qc = useQueryClient();
   808→  return useMutation({
   809→    mutationFn: (id: string) => api.toggleHideTransaction(id),
   810→    onMutate: async (id) => {
   811→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
   812→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   813→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   814→        if (!old) return old;
   815→        return old.map((t) => (t.id === id ? { ...t, isHidden: !t.isHidden } : t));
   816→      });
   817→      return { prev };
   818→    },
   819→    onError: (_e, _id, ctx) => {
   820→      if (ctx?.prev) {
   821→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   822→      }
   823→    },
   824→    onSuccess: () => {
   825→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   826→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   827→    },
   828→  });
   829→}
