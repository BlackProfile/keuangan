   532→  function handleSubmit(e: React.FormEvent) {
   533→    e.preventDefault();
   534→    setError(null);
   535→
   536→    const amt = Number(amount.replace(/[^\d.-]/g, ""));
   537→    if (!Number.isFinite(amt) || amt <= 0) {
   538→      fail("Masukkan jumlah yang valid (lebih dari 0).", "utama");
   539→      return;
   540→    }
   541→    if (!description.trim()) {
   542→      fail("Keterangan transaksi wajib diisi.", "utama");
   543→      return;
   544→    }
   545→    if (!categoryId) {
   546→      fail("Pilih kategori terlebih dahulu.", "utama");
   547→      return;
   548→    }
   549→    if (!date) {
   550→      fail("Tanggal wajib diisi.", "utama");
   551→      return;
   552→    }
   553→
   554→    let finalSplits: Array<{ amount: number; categoryId: string; note?: string }> | undefined;
   555→    if (isSplit) {
   556→      const built: Array<{ amount: number; categoryId: string; note?: string }> = [];
   557→      for (let i = 0; i < splits.length; i++) {
   558→        const s = splits[i];
   559→        const sAmt = Number(s.amount.replace(/[^\d.-]/g, ""));
   560→        if (!Number.isFinite(sAmt) || sAmt <= 0) {
   561→          fail(`Split #${i + 1}: jumlah tidak valid.`, "lainnya");
   562→          return;
   563→        }
   564→        if (!s.categoryId) {
   565→          fail(`Split #${i + 1}: pilih kategori terlebih dahulu.`, "lainnya");
   566→          return;
   567→        }
   568→        built.push({
   569→          amount: sAmt,
   570→          categoryId: s.categoryId,
   571→          note: s.note.trim() || undefined,
   572→        });
   573→      }
   574→      const total = built.reduce((sum, s) => sum + s.amount, 0);
   575→      if (Math.abs(total - amt) > 0.01) {
   576→        fail(
   577→          `Total split (${formatCurrency(total)}) harus sama dengan jumlah transaksi (${formatCurrency(amt)}).`,
   578→          "lainnya"
   579→        );
   580→        return;
   581→      }
   582→      finalSplits = built;
   583→    }
   584→
   585→    if (isDebt && !creditor.trim()) {
   586→      fail("Nama pemberi/piutang wajib diisi saat menandai hutang.", "lainnya");
   587→      return;
   588→    }
   589→
   590→    let finalReceiptItems:
   591→      | Array<{ name: string; qty: number; price: number; total: number }>
   592→      | undefined;
   593→    if (receiptItems.length > 0) {
   594→      const built: Array<{
   595→        name: string;
   596→        qty: number;
