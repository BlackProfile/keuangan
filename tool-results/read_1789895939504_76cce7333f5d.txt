    31→// Wrapper to broadcast entity changes (safe to call — no-op if socket not connected)
    32→function broadcastEntityChange(entity: string, action: string) {
    33→  try {
    34→    broadcastChange(entity, action);
    35→  } catch {
    36→    // ignore — realtime is best-effort
    37→  }
    38→}

    96→export function useCreateTransaction() {
    97→  const qc = useQueryClient();
    98→  return useMutation({
    99→    mutationFn: (data: TransactionInput) => api.createTransaction(data),
   100→    onMutate: async (data) => {
   101→      // Optimistic: add temp transaction to list cache
   102→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
   103→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   104→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   105→        if (!old) return old;
   106→        const temp = {
   107→          ...data,
   108→          id: `temp-${Date.now()}`,
   109→          createdAt: new Date().toISOString(),
   110→          updatedAt: new Date().toISOString(),
   111→          isRecurringGenerated: false,
   112→        } as Transaction;
   113→        return [temp, ...old];
   114→      });
   115→      return { prev };
   116→    },
   117→    onError: (_e, _data, ctx) => {
   118→      // Rollback on error
   119→      if (ctx?.prev) {
   120→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   121→      }
   122→    },
   123→    onSuccess: () => {
   124→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   125→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   126→      qc.invalidateQueries({ queryKey: queryKeys.analytics });
   127→      qc.invalidateQueries({ queryKey: queryKeys.budgetStatuses });
   128→      qc.invalidateQueries({ queryKey: queryKeys.accounts });
   129→      broadcastEntityChange("transaction", "create");
   130→    },
   131→  });
   132→}

   134→export function useUpdateTransaction() {
   135→  const qc = useQueryClient();
   136→  return useMutation({
   137→    mutationFn: ({ id, data }: { id: string; data: TransactionInput }) =>
   138→      api.updateTransaction(id, data),
   139→    onMutate: async ({ id, data }) => {
   140→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
   141→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   142→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   143→        if (!old) return old;
   144→        return old.map((t) => (t.id === id ? { ...t, ...data } as Transaction : t));
   145→      });
   146→      return { prev };
   147→    },
   148→    onError: (_e, _v, ctx) => {
   149→      if (ctx?.prev) {
   150→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   151→      }
   152→    },
   153→    onSuccess: () => {
   154→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   155→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   156→      qc.invalidateQueries({ queryKey: queryKeys.analytics });
   157→      qc.invalidateQueries({ queryKey: queryKeys.budgetStatuses });
   158→      qc.invalidateQueries({ queryKey: queryKeys.accounts });
   159→      broadcastEntityChange("transaction", "update");
   160→    },
   161→  });
   162→}

   164→export function useDeleteTransaction() {
   165→  const qc = useQueryClient();
   166→  return useMutation({
   167→    mutationFn: (id: string) => api.deleteTransaction(id),
   168→    onMutate: async (id) => {
   169→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
   170→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   171→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   172→        if (!old) return old;
   173→        return old.filter((t) => t.id !== id);
   174→      });
   175→      return { prev };
   176→    },
   177→    onError: (_e, _id, ctx) => {
   178→      if (ctx?.prev) {
   179→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   180→      }
   181→    },
   182→    onSuccess: () => {
   183→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   184→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   185→      qc.invalidateQueries({ queryKey: queryKeys.analytics });
   186→      qc.invalidateQueries({ queryKey: queryKeys.budgetStatuses });
   187→      qc.invalidateQueries({ queryKey: queryKeys.accounts });
   188→      broadcastEntityChange("transaction", "delete");
   189→    },
   190→  });
   191→}

   810→export function useToggleHideTransaction() {
   811→  const qc = useQueryClient();
   812→  return useMutation({
   813→    mutationFn: (id: string) => api.toggleHideTransaction(id),
   814→    onMutate: async (id) => {
   815→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
   816→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   817→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   818→        if (!old) return old;
   819→        return old.map((t) => (t.id === id ? { ...t, isHidden: !t.isHidden } : t));
   820→      });
   821→      return { prev };
   822→    },
   823→    onError: (_e, _id, ctx) => {
   824→      if (ctx?.prev) {
   825→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   826→      }
   827→    },
   828→    onSuccess: () => {
   829→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   830→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   831→    },
   832→  });
   833→}
