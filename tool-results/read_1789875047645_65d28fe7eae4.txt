   120→// ---------- Helpers ----------
   121→function parseScopeData(raw: string | null): Record<string, unknown> {
   122→  if (!raw) return {};
   123→  try {
   124→    const parsed = JSON.parse(raw);
   125→    return typeof parsed === "object" && parsed !== null
   126→      ? (parsed as Record<string, unknown>)
   127→      : {};
   128→  } catch {
   129→    return {};
   130→  }
   131→}
   132→
   133→function describeScope(
   134→  scopeType: ShareScopeType,
   135→  scopeDataStr: string | null,
   136→  ctx: {
   137→    accounts: Array<{ id: string; name: string }>;
   138→    categories: Array<{ id: string; name: string }>;
   139→    groups: Array<{ id: string; name: string }>;
   140→  }
   141→): string {
   142→  const data = parseScopeData(scopeDataStr);
   143→  switch (scopeType) {
   144→    case "ALL":
   145→      return "Semua Data";
   146→    case "ACCOUNT": {
   147→      const id = String(data.accountId ?? data.accountIds?.[0] ?? "");
   148→      const acc = ctx.accounts.find((a) => a.id === id);
   149→      return acc ? `Akun: ${acc.name}` : "Per Akun";
   150→    }
   151→    case "CATEGORY": {
   152→      const id = String(data.categoryId ?? data.categoryIds?.[0] ?? "");
   153→      const cat = ctx.categories.find((c) => c.id === id);
   154→      return cat ? `Kategori: ${cat.name}` : "Per Kategori";
   155→    }
   156→    case "GROUP": {
   157→      const id = String(data.groupId ?? data.groupIds?.[0] ?? "");
   158→      const g = ctx.groups.find((x) => x.id === id);
   159→      return g ? `Group: ${g.name}` : "Per Group";
   160→    }
   161→    case "TAG": {
   162→      const tag = String(data.tag ?? data.tags?.[0] ?? "");
   163→      return tag ? `Tag: ${tag}` : "Per Tag";
   164→    }
   165→    case "DATE_RANGE": {
   166→      const from = data.from ? formatDate(String(data.from)) : "?";
   167→      const to = data.to ? formatDate(String(data.to)) : "?";
   168→      return `Rentang: ${from} - ${to}`;
   169→    }
   170→    case "CUSTOM":
   171→      return "Pilihan Manual";
   172→    default:
   173→      return "—";
   174→  }
   175→}

   177→function statusInfo(link: ShareLink): {
   178→  label: string;
   179→  dotClass: string;
   180→  textClass: string;
   181→} {
   182→  if (!link.active) {
   183→    return {
   184→      label: "Dicabut",
   185→      dotClass: "bg-gray-400",
   186→      textClass: "text-gray-500 dark:text-gray-400",
   187→    };
   188→  }
   189→  const expired = isShareExpired({
   190→    expiresAt: link.expiresAt,
   191→    hoursActive: link.hoursActive,
   192→    createdAt: link.createdAt,
   193→    maxViews: link.maxViews,
   194→    viewCount: link.viewCount,
   195→    oneTime: link.oneTime,
   196→  });
   197→  if (expired.expired) {
   198→    return {
   199→      label: "Kadaluarsa",
   200→      dotClass: "bg-red-500",
   201→      textClass: "text-red-600 dark:text-red-400",
   202→    };
   203→  }
   204→  return {
   205→    label: "Aktif",
   206→    dotClass: "bg-emerald-500",
   207→    textClass: "text-emerald-600 dark:text-emerald-400",
   208→  };
   209→}
   210→
   211→function expiryLabel(link: ShareLink): string {
   212→  if (link.oneTime) {
   213→    if (link.viewCount >= 1) return "Sudah digunakan";
   214→    return "Sekali pakai";
   215→  }
   216→  const remaining = viewsRemaining(link);
   217→  if (remaining !== null) return `${remaining}x tersisa`;
   218→  if (link.expiresAt) {
   219→    return `Berlaku hingga ${formatDate(link.expiresAt)}`;
   220→  }
   221→  if (link.hoursActive) {
   222→    return `Aktif ${link.hoursActive} jam`;
   223→  }
   224→  return "Permanen";
   225→}

   263→      (s, l) => s + (l._count?.comments ?? 0),

   269→        expiresAt: l.expiresAt,
   270→        hoursActive: l.hoursActive,
   271→        createdAt: l.createdAt,
   272→        maxViews: l.maxViews,
   273→        viewCount: l.viewCount,
   274→        oneTime: l.oneTime,

   317→          value={stats.totalViews}

   432→// ---------- Share Card ----------
   433→function ShareCard({
   434→  share,
   435→  scopeLabel,
   436→  onEdit,
   437→  onShowQr,
   438→}: {
   439→  share: ShareLink;
   440→  scopeLabel: string;
   441→  onEdit: () => void;
   442→  onShowQr: () => void;
   443→}) {
   444→  const deleteMut = useDeleteShareLink();
   445→  const revokeMut = useRevokeShareLink();
   446→  const cloneMut = useCloneShareLink();
   447→
   448→  const status = statusInfo(share);
   449→  const access = SHARE_ACCESS_LEVELS.find((a) => a.value === share.accessLevel);
   450→  const scope = SHARE_SCOPE_TYPES.find((s) => s.value === share.scopeType);
   451→  const themeColor = share.customTheme || "#10b981";
   452→  const url = buildShareUrl(share.token);
   453→  const commentCount = share._count?.comments ?? 0;
   454→
   455→  async function handleCopy() {
   456→    try {
   457→      await navigator.clipboard.writeText(url);
   458→      toast.success("Link disalin");
   459→    } catch {
   460→      toast.error("Gagal menyalin link.");
   461→    }
   462→  }
   463→
   464→  function handleWhatsApp() {
   465→    window.open(
   466→      `https://wa.me/?text=${encodeURIComponent(url)}`,
   467→      "_blank",
   468→      "noopener,noreferrer"
   469→    );
   470→  }
   471→
   472→  function handleRevoke() {
   473→    revokeMut.mutate(share.token, {
   474→      onSuccess: () => toast.success("Link berhasil dicabut."),
   475→      onError: (err) =>
   476→        toast.error(err.message || "Gagal mencabut link."),
   477→    });
   478→  }
   479→
   480→  function handleClone() {
   481→    cloneMut.mutate(share.token, {
   482→      onSuccess: () => toast.success("Link berhasil diduplikasi."),
   483→      onError: (err) =>
   484→        toast.error(err.message || "Gagal menduplikasi link."),
   485→    });
   486→  }
   487→
   488→  function handleDelete() {
   489→    deleteMut.mutate(share.id, {
   490→      onSuccess: () => toast.success(`Link "${share.title}" dihapus.`),
   491→      onError: (err) =>
   492→        toast.error(err.message || "Gagal menghapus link."),
   493→    });
   494→  }
   495→
   496→  return (
   497→    <Card className="relative flex flex-col gap-3 p-4 transition-shadow hover:shadow-md">
   498→      {/* Header */}
   499→      <div className="flex items-start justify-between gap-3">
   500→        <div className="flex min-w-0 items-start gap-3">
   501→          <span
   502→            className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl"
   503→            style={{ backgroundColor: `${themeColor}1a` }}
   504→          >
   505→            <LucideIcon
   506→              name={access?.icon ?? "Eye"}
   507→              className="h-5 w-5"
   508→              style={{ color: themeColor }}
   509→            />
   510→          </span>
   511→          <div className="min-w-0">
   512→            <p className="truncate text-sm font-semibold text-foreground">
   513→              {share.title}
   514→            </p>
   515→            <p className="mt-0.5 text-xs text-muted-foreground">
   516→              {scopeLabel}
   517→            </p>
   518→          </div>
   519→        </div>
   520→        <Badge
   521→          variant="secondary"
   522→          className={cn(
   523→            "border-transparent",
   524→            "bg-opacity-100 text-xs font-medium"
   525→          )}
   526→          style={{
   527→            backgroundColor: `${access?.color}1a`,
   528→            color: access?.color,
   529→          }}
   530→        >
   531→          {access?.label ?? share.accessLevel}
   532→        </Badge>
   533→      </div>
   534→
   535→      {/* Stats row */}
   536→      <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-muted-foreground">
   537→        <span className="inline-flex items-center gap-1">
   538→          <Eye className="h-3.5 w-3.5" />
   539→          <span className="font-medium tabular-nums text-foreground">
   540→            {share.viewCount}
   541→          </span>{" "}
   542→          views
   543→        </span>
   544→        <span className="inline-flex items-center gap-1">
   545→          <MessageCircle className="h-3.5 w-3.5" />
   546→          <span className="font-medium tabular-nums text-foreground">
   547→            {commentCount}
   548→          </span>{" "}
   549→          komentar
   550→        </span>
   551→        {share.passwordHash && (
   552→          <span className="inline-flex items-center gap-1 text-amber-600 dark:text-amber-400">

   603→          <QrCode className="h-3.5 w-3.5" />

   609→          className="gap-1.5 flex-1"
   610→          onClick={handleWhatsApp}
   611→        >
   612→          <MessageCircle className="h-3.5 w-3.5" />
   613→          WhatsApp
   614→        </Button>
   615→
   616→        <DropdownMenu>
   617→          <DropdownMenuTrigger asChild>
   618→            <Button
   619→              size="sm"
   620→              variant="ghost"
   621→              className="h-8 w-8 p-0"
   622→              aria-label="Aksi lainnya"
   623→            >
   624→              <MoreVertical className="h-4 w-4" />
   625→            </Button>
   626→          </DropdownMenuTrigger>
