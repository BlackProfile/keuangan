    31→    31→// Wrapper to broadcast entity changes (safe to call — no-op if socket not connected)
    32→    32→function broadcastEntityChange(entity: string, action: string) {
    33→    33→  try {
    34→    34→    broadcastChange(entity, action);
    35→    35→  } catch {
    36→    36→    // ignore — realtime is best-effort
    37→    37→  }
    38→    38→}

    96→    96→export function useCreateTransaction() {
    97→    97→  const qc = useQueryClient();
    98→    98→  return useMutation({
    99→    99→    mutationFn: (data: TransactionInput) => api.createTransaction(data),
   100→   100→    onMutate: async (data) => {
   101→   101→      // Optimistic: add temp transaction to list cache
   102→   102→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
   103→   103→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   104→   104→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   105→   105→        if (!old) return old;
   106→   106→        const temp = {
   107→   107→          ...data,
   108→   108→          id: `temp-${Date.now()}`,
   109→   109→          createdAt: new Date().toISOString(),
   110→   110→          updatedAt: new Date().toISOString(),
   111→   111→          isRecurringGenerated: false,
   112→   112→        } as Transaction;
   113→   113→        return [temp, ...old];
   114→   114→      });
   115→   115→      return { prev };
   116→   116→    },
   117→   117→    onError: (_e, _data, ctx) => {
   118→   118→      // Rollback on error
   119→   119→      if (ctx?.prev) {
   120→   120→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   121→   121→      }
   122→   122→    },
   123→   123→    onSuccess: () => {
   124→   124→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   125→   125→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   126→   126→      qc.invalidateQueries({ queryKey: queryKeys.analytics });
   127→   127→      qc.invalidateQueries({ queryKey: queryKeys.budgetStatuses });
   128→   128→      qc.invalidateQueries({ queryKey: queryKeys.accounts });
   129→   129→      broadcastEntityChange("transaction", "create");
   130→   130→    },
   131→   131→  });
   132→   132→}

   134→   134→export function useUpdateTransaction() {
   135→   135→  const qc = useQueryClient();
   136→   136→  return useMutation({
   137→   137→    mutationFn: ({ id, data }: { id: string; data: TransactionInput }) =>
   138→   138→      api.updateTransaction(id, data),
   139→   139→    onMutate: async ({ id, data }) => {
   140→   140→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
   141→   141→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   142→   142→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   143→   143→        if (!old) return old;
   144→   144→        return old.map((t) => (t.id === id ? { ...t, ...data } as Transaction : t));
   145→   145→      });
   146→   146→      return { prev };
   147→   147→    },
   148→   148→    onError: (_e, _v, ctx) => {
   149→   149→      if (ctx?.prev) {
   150→   150→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   151→   151→      }
   152→   152→    },
   153→   153→    onSuccess: () => {
   154→   154→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   155→   155→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   156→   156→      qc.invalidateQueries({ queryKey: queryKeys.analytics });
   157→   157→      qc.invalidateQueries({ queryKey: queryKeys.budgetStatuses });
   158→   158→      qc.invalidateQueries({ queryKey: queryKeys.accounts });
   159→   159→      broadcastEntityChange("transaction", "update");
   160→   160→    },
   161→   161→  });

   177→   177→    onError: (_e, _id, ctx) => {
   178→   178→      if (ctx?.prev) {

   180→   180→      }
   181→   181→    },
   182→   182→    onSuccess: () => {
   183→   183→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   184→   184→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   185→   185→      qc.invalidateQueries({ queryKey: queryKeys.analytics });
   186→   186→      qc.invalidateQueries({ queryKey: queryKeys.budgetStatuses });
   187→   187→      qc.invalidateQueries({ queryKey: queryKeys.accounts });

   810→   810→export function useToggleHideTransaction() {
   811→   811→  const qc = useQueryClient();
   812→   812→  return useMutation({
   813→   813→    mutationFn: (id: string) => api.toggleHideTransaction(id),
   814→   814→    onMutate: async (id) => {
   815→   815→      await qc.cancelQueries({ queryKey: queryKeys.transactions });
   816→   816→      const prev = qc.getQueriesData<Transaction[]>({ queryKey: queryKeys.transactions });
   817→   817→      qc.setQueriesData<Transaction[]>({ queryKey: queryKeys.transactions }, (old) => {
   818→   818→        if (!old) return old;
   819→   819→        return old.map((t) => (t.id === id ? { ...t, isHidden: !t.isHidden } : t));
   820→   820→      });
   821→   821→      return { prev };
   822→   822→    },
   823→   823→    onError: (_e, _id, ctx) => {
   824→   824→      if (ctx?.prev) {
   825→   825→        ctx.prev.forEach(([key, data]) => qc.setQueryData(key, data));
   826→   826→      }
   827→   827→    },
   828→   828→    onSuccess: () => {
   829→   829→      qc.invalidateQueries({ queryKey: queryKeys.transactions });
   830→   830→      qc.invalidateQueries({ queryKey: queryKeys.dashboard });
   831→   831→    },
   832→   832→  });
   833→   833→}
