   397→  async function downloadExport(fmt: DownloadFormat) {
   398→    const url: Record<DownloadFormat, string> = {
   399→      pdf: api.exportPdfUrl(),
   400→      excel: api.exportExcelUrl(),
   401→      csv: api.exportCsvUrl(),
   402→      json: api.exportJsonUrl(),
   403→    };
   404→    try {
   405→      setDownloading(fmt);
   406→      const res = await fetch(url[fmt], {
   407→        method: "POST",
   408→        headers: { "Content-Type": "application/json" },
   409→        body: JSON.stringify({ scope, fields, options }),
   410→      });
   411→      if (!res.ok) throw new Error("Export gagal");
   412→      const blob = await res.blob();
   413→      const a = document.createElement("a");
   414→      a.href = URL.createObjectURL(blob);
   415→      a.download = `dompetku-export-${new Date()
   416→        .toISOString()
   417→        .split("T")[0]}.${fmt === "excel" ? "xlsx" : fmt}`;
   418→      document.body.appendChild(a);
   419→      a.click();
   420→      document.body.removeChild(a);
   421→      URL.revokeObjectURL(a.href);
   422→      toast.success(`Export ${fmt.toUpperCase()} berhasil`);
   423→    } catch (err) {
   424→      const msg = err instanceof Error ? err.message : "Export gagal";
   425→      toast.error(msg);
   426→    } finally {
   427→      setDownloading(null);
   428→    }
   429→  }
